pub fn from_fn<DB: Database>(
    up: impl for<'a> Fn(&'a mut <DB as Database>::Connection) -> OperationFuture<'a> + Send + Sync + 'static,
    down: impl for<'a> Fn(&'a mut <DB as Database>::Connection) -> OperationFuture<'a> + Send + Sync + 'static,
) -> FnOperation<DB>
Expand description

Builds a reversible operation from two functions which get the connection and return a boxed future, usually written as Box::pin(async move { .. }). The database has to be named since the closures cannot tell it, for example from_fn::<Postgres>(..).

To run more than one query pass &mut *connection to each.

§Example

use sqlx::Sqlite;
use sqlx_migrator::operation::from_fn;

let operation = from_fn::<Sqlite>(
    |connection| {
        Box::pin(async move {
            sqlx::query("CREATE TABLE sample (id INTEGER)")
                .execute(connection)
                .await?;
            Ok(())
        })
    },
    |connection| {
        Box::pin(async move {
            sqlx::query("DROP TABLE sample").execute(connection).await?;
            Ok(())
        })
    },
);
