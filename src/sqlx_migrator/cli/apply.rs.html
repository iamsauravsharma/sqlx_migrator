1//! `apply` subcommand
2#![expect(clippy::print_stdout, reason = "allow printing to stdout in cli")]
3use clap::Parser;
4use sqlx::Database;
5
6use super::output::{OutputFormat, confirm, render_plan};
7use crate::error::Error;
8use crate::migrator::{Migrator, Plan, SupportedDatabase};
9
10#[derive(Parser, Debug)]
11#[expect(clippy::struct_excessive_bools)]
12pub(super) struct Apply {
13    /// App name up to which migration needs to be applied. If migration option
14    /// is also present then only till migration is applied
15    #[arg(long)]
16    app: Option<String>,
17    /// Check for pending migration
18    #[arg(long, conflicts_with_all = ["count", "fake", "force", "plan"])]
19    check: bool,
20    /// Number of migration to apply. Conflicts with app args
21    #[arg(long, conflicts_with = "app")]
22    count: Option<usize>,
23    /// Make migration applied without running migration operations
24    #[arg(long)]
25    fake: bool,
26    /// Force run apply operation without asking question if migration is
27    /// destructible
28    #[arg(long)]
29    force: bool,
30    /// Output format of the plan. Requires plan option to be present
31    #[arg(long, value_enum, default_value_t, requires = "plan")]
32    pub(super) format: OutputFormat,
33    /// Apply migration till provided migration. Requires app options to be
34    /// present
35    #[arg(long, requires = "app")]
36    migration: Option<String>,
37    /// Show plan
38    #[arg(long, conflicts_with_all = ["fake", "force"])]
39    pub(super) plan: bool,
40}
41
42impl Apply {
43    pub(super) async fn run<DB>(
44        &self,
45        connection: &mut <DB as Database>::Connection,
46        migrator: &Migrator<DB>,
47    ) -> Result<(), Error>
48    where
49        DB: SupportedDatabase,
50    {
51        let plan = if let Some(count) = self.count {
52            Plan::apply_count(count)
53        } else if let Some(app) = &self.app {
54            Plan::apply_name(app, self.migration.as_deref())
55        } else {
56            Plan::apply_all()
57        }
58        .fake(self.fake);
59        let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
60        let migrations = migrator.generate_migration_plan(Some(&plan), &applied_rows)?;
61        if self.check {
62            if !migrations.is_empty() {
63                return Err(Error::PendingMigrationPresent);
64            }
65            println!("No pending migration to apply");
66            return Ok(());
67        }
68        if self.plan {
69            println!(
70                "{}",
71                render_plan(
72                    migrations.iter().map(AsRef::as_ref),
73                    self.format,
74                    "No migration exists for applying"
75                )?
76            );
77            return Ok(());
78        }
79        let destructible_migrations = migrations
80            .iter()
81            .filter(|m| m.operations().iter().any(|o| o.is_destructible()))
82            .map(|m| (m.app(), m.name()))
83            .collect::<Vec<_>>();
84        if !self.force
85            && !self.fake
86            && !destructible_migrations.is_empty()
87            && !confirm(
88                &format!(
89                    "Do you want to apply {} destructible migrations?",
90                    destructible_migrations.len()
91                ),
92                &destructible_migrations,
93            )?
94        {
95            println!("Aborted applying migrations");
96            return Ok(());
97        }
98        let report = migrator.run(connection, &plan).await?;
99        println!(
100            "Successfully applied {} migrations according to plan in {}ms",
101            report.len(),
102            report.total_elapsed().as_millis()
103        );
104        Ok(())
105    }
106}
