1//! `revert` subcommand
2#![expect(clippy::print_stdout, reason = "allow printing to stdout in cli")]
3use clap::Parser;
4use sqlx::Database;
5
6use super::output::{OutputFormat, confirm, render_plan};
7use crate::error::Error;
8use crate::migrator::{Migrator, Plan, SupportedDatabase};
9
10#[derive(Parser, Debug)]
11#[expect(clippy::struct_excessive_bools)]
12pub(super) struct Revert {
13    /// Revert all migration. Conflicts with app args
14    #[arg(long, conflicts_with = "app")]
15    all: bool,
16    /// Revert migration till app migrations is reverted. If it is present
17    /// alongside migration options then only till migration is reverted
18    #[arg(long)]
19    app: Option<String>,
20    /// Number of migration to revert. Conflicts with all and app args
21    #[arg(long, conflicts_with_all = ["all", "app"])]
22    count: Option<usize>,
23    /// Revert the migrations applied by the last run, which share a batch.
24    /// Conflicts with all, app and count args
25    #[arg(long, conflicts_with_all = ["all", "app", "count"])]
26    last_batch: bool,
27    /// Make migration reverted without running revert operation
28    #[arg(long)]
29    fake: bool,
30    /// Run the whole plan in one transaction, so it is reverted entirely or not
31    /// at all. Every migration has to be atomic
32    #[arg(long, conflicts_with_all = ["plan"])]
33    grouped: bool,
34    /// Force run revert operation without asking question
35    #[arg(long)]
36    force: bool,
37    /// Output format of the plan. Requires plan option to be present
38    #[arg(long, value_enum, default_value_t, requires = "plan")]
39    format: OutputFormat,
40    /// Revert migration till provided migration. Requires app options to be
41    /// present
42    #[arg(long, requires = "app")]
43    migration: Option<String>,
44    /// Show plan
45    #[arg(long, conflicts_with_all = ["fake", "force"])]
46    plan: bool,
47}
48
49impl Revert {
50    pub(super) async fn run<DB>(
51        &self,
52        connection: &mut <DB as Database>::Connection,
53        migrator: &Migrator<DB>,
54    ) -> Result<(), Error>
55    where
56        DB: SupportedDatabase,
57    {
58        let mut revert_last = false;
59        let plan = if let Some(count) = self.count {
60            Plan::revert_count(count)
61        } else if let Some(app) = &self.app {
62            Plan::revert_name(app, self.migration.as_deref())
63        } else if self.all {
64            Plan::revert_all()
65        } else if self.last_batch {
66            Plan::revert_last_batch()
67        } else {
68            revert_last = true;
69            Plan::revert_count(1)
70        }
71        .fake(self.fake)
72        .grouped(self.grouped);
73        let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
74        // reverting the last migration when nothing is applied is not an
75        // error, unlike an explicit count which is larger than what is applied
76        let nothing_applied = revert_last
77            && migrator
78                .generate_migration_plan(Some(&Plan::revert_all()), &applied_rows)?
79                .is_empty();
80        let revert_migrations = if nothing_applied {
81            vec![]
82        } else {
83            migrator.generate_migration_plan(Some(&plan), &applied_rows)?
84        };
85        if self.plan {
86            println!(
87                "{}",
88                render_plan(
89                    revert_migrations.iter().map(AsRef::as_ref),
90                    self.format,
91                    "No migration exists for reverting"
92                )?
93            );
94            return Ok(());
95        }
96        if revert_migrations.is_empty() {
97            println!("No migration exists for reverting");
98            return Ok(());
99        }
100        let items = revert_migrations
101            .iter()
102            .map(|m| (m.app(), m.name()))
103            .collect::<Vec<_>>();
104        if !self.force
105            && !self.fake
106            && !items.is_empty()
107            && !confirm(
108                &format!("Do you want to revert {} migrations?", items.len()),
109                &items,
110            )?
111        {
112            println!("Aborted reverting migrations");
113            return Ok(());
114        }
115        let report = migrator.run(connection, &plan).await?;
116        println!(
117            "Successfully reverted {} migrations according to plan in {}ms",
118            report.len(),
119            report.total_elapsed().as_millis()
120        );
121        Ok(())
122    }
123}
