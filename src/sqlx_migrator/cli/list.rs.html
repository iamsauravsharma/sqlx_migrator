16/// Status of a migration as shown by `list`
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18#[cfg_attr(feature = "json", derive(Serialize), serde(rename_all = "lowercase"))]
19pub(super) enum ListStatus {
20    /// Applied and recorded in the migration table
21    Applied,
22    /// Started running and never finished, so the database may be partly
23    /// migrated. Fix it by hand and run the `repair` command
24    Failed,
25    /// Known migration which is not applied yet
26    Pending,
27    /// Known migration which will not be applied because a migration replacing
28    /// it is applied or will be applied instead
29    Replaced,
30    /// Recorded in the migration table but no longer present in the migration
31    /// list. Such entries can be removed with the `prune` command
32    Unknown,
33}
34
35impl ListStatus {
36    fn symbol(self) -> &'static str {
37        match self {
38            ListStatus::Applied => "\u{2713}",
39            ListStatus::Failed => "!",
40            ListStatus::Pending => "\u{2717}",
41            ListStatus::Replaced => "\u{2194}",
42            ListStatus::Unknown => "?",
43        }
44    }
45}

47/// A row of the `list` command
48#[cfg_attr(feature = "json", derive(Serialize))]
49pub(super) struct ListEntry<'a> {
50    pub(super) id: Option<i32>,
51    pub(super) app: &'a str,
52    pub(super) name: &'a str,
53    pub(super) status: ListStatus,
54    pub(super) applied_time: Option<String>,
55    pub(super) execution_time_ns: Option<i64>,
56    pub(super) batch: Option<i32>,
57}
58
59/// Build the list entries: every known migration in apply order followed by
60/// applied rows which are not known any more.
61pub(super) fn list_entries<'a, DB>(
62    migrator: &'a Migrator<DB>,
63    applied_migrations: &'a [AppliedMigrationSqlRow],
64) -> Result<Vec<ListEntry<'a>>, Error>
65where
66    DB: SupportedDatabase,
67{
68    let status_list = migrator.status(applied_migrations)?;
69    let apply_plan_keys = if status_list.is_empty() {
70        HashSet::new()
71    } else {
72        migrator
73            .generate_migration_plan(Some(&Plan::apply_all()), applied_migrations)?
74            .iter()
75            .map(|migration| (migration.app(), migration.name()))
76            .collect::<HashSet<_>>()
77    };
78    let known_keys = status_list
79        .iter()
80        .map(|(migration, _)| (migration.app(), migration.name()))
81        .collect::<HashSet<_>>();
82    let mut entries = status_list
83        .iter()
84        .map(|(migration, applied)| {
85            let key = (migration.app(), migration.name());
86            match applied {
87                Some(row) => {
88                    ListEntry {
89                        id: Some(row.id()),
90                        app: migration.app(),
91                        name: migration.name(),
92                        status: if row.success() {
93                            ListStatus::Applied
94                        } else {
95                            ListStatus::Failed
96                        },
97                        applied_time: Some(format_time(row.applied_time())),
98                        execution_time_ns: Some(row.execution_time_ns()),
99                        batch: Some(row.batch()),
100                    }
101                }
102                None => {
103                    ListEntry {
104                        id: None,
105                        app: migration.app(),
106                        name: migration.name(),
107                        status: if apply_plan_keys.contains(&key) {
108                            ListStatus::Pending
109                        } else {
110                            ListStatus::Replaced
111                        },
112                        applied_time: None,
113                        execution_time_ns: None,
114                        batch: None,
115                    }
116                }
117            }
118        })
119        .collect::<Vec<_>>();
120    // applied migrations which no longer exist in the migration list (e.g.
121    // migration was removed from the codebase) so drift is visible
122    entries.extend(
123        applied_migrations
124            .iter()
125            .filter(|row| !known_keys.contains(&(row.app(), row.name())))
126            .map(|row| {
127                ListEntry {
128                    id: Some(row.id()),
129                    app: row.app(),
130                    name: row.name(),
131                    status: ListStatus::Unknown,
132                    applied_time: Some(format_time(row.applied_time())),
133                    execution_time_ns: Some(row.execution_time_ns()),
134                    batch: Some(row.batch()),
135                }
136            }),
137    );
138    Ok(entries)
139}

141/// Applied time in UTC as `YYYY-MM-DD HH:MM:SS`
142fn format_time(applied_time: OffsetDateTime) -> String {
143    applied_time
144        .to_offset(UtcOffset::UTC)
145        .format(time::macros::format_description!(
146            version = 2,
147            "[year]-[month]-[day] [hour]:[minute]:[second]"
148        ))
149        .unwrap_or_default()
150}
151
152/// Time kept in nanoseconds as milliseconds with a fraction, for example
153/// `1.234` for 1234567 nanoseconds
154pub(super) fn milliseconds(nanoseconds: i64) -> String {
155    format!(
156        "{}.{:03}",
157        nanoseconds / 1_000_000,
158        nanoseconds % 1_000_000 / 1_000
159    )
160}

162/// Render list entries in the requested format
163#[cfg_attr(
164    not(feature = "json"),
165    expect(
166        clippy::unnecessary_wraps,
167        reason = "only JSON serialization can fail; keep one signature"
168    )
169)]
170pub(super) fn render_list(
171    entries: &[ListEntry<'_>],
172    format: OutputFormat,
173) -> Result<String, Error> {
174    match format {
175        #[cfg(feature = "json")]
176        OutputFormat::Json => Ok(serde_json::to_string_pretty(entries)?),
177        OutputFormat::Table => {
178            let rows = entries
179                .iter()
180                .map(|entry| {
181                    vec![
182                        entry
183                            .id
184                            .map_or_else(|| "N/A".to_string(), |id| id.to_string()),
185                        entry.app.to_string(),
186                        entry.name.to_string(),
187                        entry.status.symbol().to_string(),
188                        entry
189                            .applied_time
190                            .clone()
191                            .unwrap_or_else(|| "N/A".to_string()),
192                        entry
193                            .batch
194                            .map_or_else(|| "N/A".to_string(), |batch| batch.to_string()),
195                        entry
196                            .execution_time_ns
197                            .map_or_else(|| "N/A".to_string(), milliseconds),
198                    ]
199                })
200                .collect::<Vec<_>>();
201            Ok(render_table(
202                &[
203                    "ID",
204                    "App",
205                    "Name",
206                    "Status",
207                    "Applied time",
208                    "Batch",
209                    "Time (ms)",
210                ],
211                &rows,
212            ))
213        }
214    }
215}

217#[derive(Parser, Debug)]
218pub(super) struct List {
219    /// Output format
220    #[arg(long, value_enum, default_value_t)]
221    pub(super) format: OutputFormat,
222}
223
224impl List {
225    pub(super) async fn run<DB>(
226        &self,
227        connection: &mut <DB as Database>::Connection,
228        migrator: &Migrator<DB>,
229    ) -> Result<(), Error>
230    where
231        DB: SupportedDatabase,
232    {
233        let applied_migrations = migrator.fetch_applied_migration_from_db(connection).await?;
234        let entries = list_entries(migrator, &applied_migrations)?;
235        println!("{}", render_list(&entries, self.format)?);
236        Ok(())
237    }
