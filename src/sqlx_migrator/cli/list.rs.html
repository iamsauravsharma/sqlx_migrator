15/// Status of a migration as shown by `list`
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17#[cfg_attr(feature = "json", derive(Serialize), serde(rename_all = "lowercase"))]
18pub(super) enum ListStatus {
19    /// Applied and recorded in the migration table
20    Applied,
21    /// Known migration which is not applied yet
22    Pending,
23    /// Known migration which will not be applied because a migration replacing
24    /// it is applied or will be applied instead
25    Replaced,
26    /// Recorded in the migration table but no longer present in the migration
27    /// list. Such entries can be removed with the `prune` command
28    Unknown,
29}
30
31impl ListStatus {
32    fn symbol(self) -> &'static str {
33        match self {
34            ListStatus::Applied => "\u{2713}",
35            ListStatus::Pending => "\u{2717}",
36            ListStatus::Replaced => "\u{2194}",
37            ListStatus::Unknown => "?",
38        }
39    }
40}

42/// A row of the `list` command
43#[cfg_attr(feature = "json", derive(Serialize))]
44pub(super) struct ListEntry<'a> {
45    pub(super) id: Option<i32>,
46    pub(super) app: &'a str,
47    pub(super) name: &'a str,
48    pub(super) status: ListStatus,
49    pub(super) applied_time: Option<&'a str>,
50}
51
52/// Build the list entries: every known migration in apply order followed by
53/// applied rows which are not known any more.
54pub(super) fn list_entries<'a, DB, M>(
55    migrator: &'a M,
56    applied_migrations: &'a [AppliedMigrationSqlRow],
57) -> Result<Vec<ListEntry<'a>>, Error>
58where
59    DB: Database,
60    M: Migrate<DB> + ?Sized,
61{
62    let status_list = migrator.status(applied_migrations)?;
63    let apply_plan_keys = if status_list.is_empty() {
64        HashSet::new()
65    } else {
66        migrator
67            .generate_migration_plan(Some(&Plan::apply_all()), applied_migrations)?
68            .iter()
69            .map(|migration| (migration.app(), migration.name()))
70            .collect::<HashSet<_>>()
71    };
72    let known_keys = status_list
73        .iter()
74        .map(|(migration, _)| (migration.app(), migration.name()))
75        .collect::<HashSet<_>>();
76    let mut entries = status_list
77        .iter()
78        .map(|(migration, applied)| {
79            let key = (migration.app(), migration.name());
80            match applied {
81                Some(row) => {
82                    ListEntry {
83                        id: Some(row.id()),
84                        app: migration.app(),
85                        name: migration.name(),
86                        status: ListStatus::Applied,
87                        applied_time: Some(row.applied_time()),
88                    }
89                }
90                None => {
91                    ListEntry {
92                        id: None,
93                        app: migration.app(),
94                        name: migration.name(),
95                        status: if apply_plan_keys.contains(&key) {
96                            ListStatus::Pending
97                        } else {
98                            ListStatus::Replaced
99                        },
100                        applied_time: None,
101                    }
102                }
103            }
104        })
105        .collect::<Vec<_>>();
106    // applied migrations which no longer exist in the migration list (e.g.
107    // migration was removed from the codebase) so drift is visible
108    entries.extend(
109        applied_migrations
110            .iter()
111            .filter(|row| !known_keys.contains(&(row.app(), row.name())))
112            .map(|row| {
113                ListEntry {
114                    id: Some(row.id()),
115                    app: row.app(),
116                    name: row.name(),
117                    status: ListStatus::Unknown,
118                    applied_time: Some(row.applied_time()),
119                }
120            }),
121    );
122    Ok(entries)
123}

125/// Render list entries in the requested format
126#[cfg_attr(
127    not(feature = "json"),
128    expect(
129        clippy::unnecessary_wraps,
130        reason = "only JSON serialization can fail; keep one signature"
131    )
132)]
133pub(super) fn render_list(
134    entries: &[ListEntry<'_>],
135    format: OutputFormat,
136) -> Result<String, Error> {
137    match format {
138        #[cfg(feature = "json")]
139        OutputFormat::Json => Ok(serde_json::to_string_pretty(entries)?),
140        OutputFormat::Table => {
141            let rows = entries
142                .iter()
143                .map(|entry| {
144                    vec![
145                        entry
146                            .id
147                            .map_or_else(|| "N/A".to_string(), |id| id.to_string()),
148                        entry.app.to_string(),
149                        entry.name.to_string(),
150                        entry.status.symbol().to_string(),
151                        entry.applied_time.unwrap_or("N/A").to_string(),
152                    ]
153                })
154                .collect::<Vec<_>>();
155            Ok(render_table(
156                &["ID", "App", "Name", "Status", "Applied time"],
157                &rows,
158            ))
159        }
160    }
161}

163#[derive(Parser, Debug)]
164pub(super) struct List {
165    /// Output format
166    #[arg(long, value_enum, default_value_t)]
167    pub(super) format: OutputFormat,
168}
169
170impl List {
171    pub(super) async fn run<DB, M>(
172        &self,
173        connection: &mut <DB as Database>::Connection,
174        migrator: &M,
175    ) -> Result<(), Error>
176    where
177        DB: Database,
178        M: Migrate<DB> + ?Sized,
179    {
180        migrator.ensure_migration_table_exists(connection).await?;
181        let applied_migrations = migrator.fetch_applied_migration_from_db(connection).await?;
182        let entries = list_entries(migrator, &applied_migrations)?;
183        println!("{}", render_list(&entries, self.format)?);
184        Ok(())
185    }
186}
