1//! `redo` subcommand
2#![expect(clippy::print_stdout, reason = "allow printing to stdout in cli")]
3use clap::Parser;
4use sqlx::Database;
5
6use super::output::confirm;
7use crate::error::Error;
8use crate::migrator::{Migrator, Plan, SupportedDatabase};
9
10#[derive(Parser, Debug)]
11pub(super) struct Redo {
12    /// Redo without asking for confirmation
13    #[arg(long)]
14    pub(super) force: bool,
15}
16
17impl Redo {
18    pub(super) async fn run<DB>(
19        &self,
20        connection: &mut <DB as Database>::Connection,
21        migrator: &Migrator<DB>,
22    ) -> Result<(), Error>
23    where
24        DB: SupportedDatabase,
25    {
26        let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
27        // the revert all plan lists applied migrations last applied first,
28        // so its head is the migration to redo
29        let applied = migrator.generate_migration_plan(Some(&Plan::revert_all()), &applied_rows)?;
30        let Some((app, name)) = applied
31            .first()
32            .map(|m| (m.app().to_string(), m.name().to_string()))
33        else {
34            println!("No applied migration to redo");
35            return Ok(());
36        };
37        let revert_plan = Plan::revert_count(1);
38        if !self.force
39            && !confirm(
40                "Do you want to revert and apply again the last applied migration?",
41                &[(&app, &name)],
42            )?
43        {
44            println!("Aborted redoing migration");
45            return Ok(());
46        }
47        migrator.run(connection, &revert_plan).await?;
48        let report = migrator
49            .run(connection, &Plan::apply_name(&app, Some(&name)))
50            .await?;
51        println!(
52            "Successfully redid {app} : {name} in {}ms",
53            report.total_elapsed().as_millis()
54        );
55        Ok(())
56    }
57}
