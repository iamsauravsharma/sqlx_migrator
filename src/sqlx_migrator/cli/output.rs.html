11/// Output format of commands which print lists
12#[derive(ValueEnum, Clone, Copy, Debug, Default, PartialEq, Eq)]
13pub(super) enum OutputFormat {
14    /// Human readable table
15    #[default]
16    Table,
17    /// JSON array, one object per migration
18    Json,
19}

21/// Render rows as a table whose column widths fit the content. Cells are
22/// centered and columns separated by ` | `, with a dashed line below the
23/// header.
24pub(super) fn render_table(headers: &[&str], rows: &[Vec<String>]) -> String {
25    let widths = headers
26        .iter()
27        .enumerate()
28        .map(|(column, header)| {
29            rows.iter()
30                .map(|row| row.get(column).map_or(0, |cell| cell.chars().count()))
31                .chain(std::iter::once(header.chars().count()))
32                .max()
33                .unwrap_or_default()
34        })
35        .collect::<Vec<_>>();
36    let render_row = |cells: &[&str]| {
37        cells
38            .iter()
39            .zip(&widths)
40            .map(|(cell, width)| format!("{cell:^width$}"))
41            .collect::<Vec<_>>()
42            .join(" | ")
43    };
44    let full_width = widths.iter().sum::<usize>() + widths.len().saturating_sub(1) * 3;
45    let mut output = render_row(headers);
46    output.push('\n');
47    output.push_str(&"-".repeat(full_width));
48    for row in rows {
49        output.push('\n');
50        let cells = row.iter().map(String::as_str).collect::<Vec<_>>();
51        output.push_str(&render_row(&cells));
52    }
53    output
54}

56/// A migration in a plan, as printed by `apply --plan` and `revert --plan`
57#[derive(Serialize)]
58struct PlanEntry<'a> {
59    app: &'a str,
60    name: &'a str,
61}
62
63/// Render a plan in the requested format. `empty_message` is used for the
64/// table format when the plan has no migration.
65pub(super) fn render_plan<'migration, DB>(
66    migrations: impl IntoIterator<Item = &'migration dyn Migration<DB>>,
67    format: OutputFormat,
68    empty_message: &str,
69) -> Result<String, Error>
70where
71    DB: 'migration,
72{
73    let entries = migrations
74        .into_iter()
75        .map(|migration| {
76            PlanEntry {
77                app: migration.app(),
78                name: migration.name(),
79            }
80        })
81        .collect::<Vec<_>>();
82    match format {
83        OutputFormat::Json => Ok(serde_json::to_string_pretty(&entries)?),
84        OutputFormat::Table if entries.is_empty() => Ok(empty_message.to_string()),
85        OutputFormat::Table => {
86            let rows = entries
87                .iter()
88                .map(|entry| vec![entry.app.to_string(), entry.name.to_string()])
89                .collect::<Vec<_>>();
90            Ok(render_table(&["App", "Name"], &rows))
91        }
92    }
93}

95/// Ask the user to confirm an action listing the affected migrations. Returns
96/// `Ok(false)` when the user declines.
97///
98/// # Errors
99/// When stdin is not a terminal so no answer can be read
100pub(super) fn confirm(question: &str, items: &[(&str, &str)]) -> Result<bool, Error> {
101    if !std::io::stdin().is_terminal() {
102        return Err(Error::ConfirmationRequired);
103    }
104    println!("{question} (y/N)");
105    for (position, (app, name)) in items.iter().enumerate() {
106        println!("{}. {app} : {name}", position + 1);
107    }
108    std::io::stdout().flush()?;
109    let mut input = String::new();
110    std::io::stdin().read_line(&mut input)?;
111    let input_trimmed = input.trim().to_ascii_lowercase();
112    Ok(["y", "yes"].contains(&input_trimmed.as_str()))
113}
