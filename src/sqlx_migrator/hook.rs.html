1//! Hooks run functions at fixed points of a run
2//!
3//! Register a function with [`Migrator::add_hook`](crate::Migrator::add_hook)
4//! and the [`HookType`] saying when it runs. It gets the connection doing the
5//! work, which is the transaction of an atomic migration, so a hook can set
6//! transaction scoped options such as Postgres `SET LOCAL lock_timeout`. When
7//! the lock strategy holds the whole plan in one transaction every hook runs
8//! inside it. A hook which fails stops the run and rolls back the transaction
9//! it ran in.
10//!
11//! Migration hooks are not called for a fake plan since it executes nothing.
12#![cfg_attr(
13    feature = "postgres",
14    doc = r#"
15# Example
16Give up on a lock held by another session after five seconds instead of
17waiting for ever
18```rust,no_run
19use sqlx::{Executor as _, Postgres};
20use sqlx_migrator::hook::{HookContext, HookFuture, HookType};
21use sqlx_migrator::Migrator;
22
23fn lock_timeout<'a>(
24    connection: &'a mut sqlx::PgConnection,
25    _context: &'a HookContext<'a, Postgres>,
26) -> HookFuture<'a> {
27    Box::pin(async move {
28        connection.execute("SET LOCAL lock_timeout = '5s'").await?;
29        Ok(())
30    })
31}
32
33let mut migrator = Migrator::<Postgres>::default();
34migrator.add_hook(HookType::BeforeMigration, lock_timeout);
35// a closure works the same way
36migrator.add_hook(HookType::AfterRun, |_connection, context| {
37    Box::pin(async move {
38        println!("ran {} migrations", context.report().map_or(0, |report| report.len()));
39        Ok(())
40    })
41});
42```
43"#

46use std::pin::Pin;
47
48use sqlx::Database;
49
50use crate::error::Error;
51use crate::migration::Migration;
52use crate::migrator::RunReport;
53
54/// Future returned by a hook function
55pub type HookFuture<'a> = Pin<Box<dyn Future<Output = Result<(), Error>> + Send + 'a>>;
56
57pub(crate) type BoxHook<DB> = Box<
58    dyn for<'a> Fn(&'a mut <DB as Database>::Connection, &'a HookContext<'a, DB>) -> HookFuture<'a>
59        + Send
60        + Sync,
61>;
62
63/// When a hook function runs
64#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
65#[non_exhaustive]
66pub enum HookType {
67    /// Once the lock is taken, before the first migration
68    BeforeRun,
69    /// After the last migration, before the lock is released. The context has
70    /// the [`RunReport`]
71    AfterRun,
72    /// Before the operations of a migration, inside its transaction when it is
73    /// atomic. The context has the migration
74    BeforeMigration,
75    /// After the operations of a migration, inside its transaction when it is
76    /// atomic, so a failure rolls the migration back. The context has the
77    /// migration
78    AfterMigration,
79}
80
81/// What a hook function is told about the run
82pub struct HookContext<'a, DB> {
83    apply: bool,
84    migration: Option<&'a dyn Migration<DB>>,
85    report: Option<&'a RunReport>,
86}
87
88impl<'a, DB> HookContext<'a, DB> {
89    pub(crate) fn new(
90        apply: bool,
91        migration: Option<&'a dyn Migration<DB>>,
92        report: Option<&'a RunReport>,
93    ) -> Self {
94        Self {
95            apply,
96            migration,
97            report,
98        }
99    }
100
101    /// `true` when migrations are applied and `false` when they are reverted
102    #[must_use]
103    pub fn is_apply(&self) -> bool {
104        self.apply
105    }
106
107    /// The migration for [`HookType::BeforeMigration`] and
108    /// [`HookType::AfterMigration`], `None` otherwise
109    #[must_use]
110    pub fn migration(&self) -> Option<&'a dyn Migration<DB>> {
111        self.migration
112    }
113
114    /// The report for [`HookType::AfterRun`], `None` otherwise
115    #[must_use]
116    pub fn report(&self) -> Option<&'a RunReport> {
117        self.report
118    }
119}
