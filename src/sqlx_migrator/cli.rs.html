1//! Module for creating and running cli with help of migrator
2//!
3//! CLI Command can directly used or extended
4//!
5//! For direct usage you can run `parse_and_run` function for `MigrationCommand`
6//!
7//! OR
8//!
9//! If you want to extend your own clap based cli then you can add migrator to
10//! sub command enum and then run migrator
11//! ```rust,no_run
12//! #[derive(clap::Parser)]
13//! struct Cli {
14//!     #[command(subcommand)]
15//!     sub_command: CliSubcommand,
16//! }
17//!
18//! #[derive(clap::Subcommand)]
19//! enum CliSubcommand {
20//!     #[command()]
21//!     Migrator(sqlx_migrator::cli::MigrationCommand),
22//! }
23//! ```
24#![expect(clippy::print_stdout, reason = "allow printing to stdout in cli")]

34/// Migration command for performing rust based sqlx migrations
35#[derive(Parser, Debug)]
36pub struct MigrationCommand {
37    #[command(subcommand)]
38    sub_command: SubCommand,
39}
40
41impl MigrationCommand {
42    /// Parse [`MigrationCommand`] and run migration command line interface
43    ///
44    /// # Errors
45    /// If migration command fails to complete and raise some issue
46    pub async fn parse_and_run<DB, M>(
47        connection: &mut <DB as Database>::Connection,
48        migrator: &M,
49    ) -> Result<(), Error>
50    where
51        DB: Database,
52        M: Migrate<DB> + ?Sized,
53    {
54        let migration_command = Self::parse();
55        migration_command.run(connection, migrator).await
56    }
57
58    /// Run migration command line interface
59    ///
60    /// # Errors
61    /// If migration command fails to complete and raise some issue
62    pub async fn run<DB, M>(
63        &self,
64        connection: &mut <DB as Database>::Connection,
65        migrator: &M,
66    ) -> Result<(), Error>
67    where
68        DB: Database,
69        M: Migrate<DB> + ?Sized,
70    {
71        self.sub_command
72            .handle_subcommand(migrator, connection)
73            .await?;
74        Ok(())
75    }
76}
77
78#[derive(Subcommand, Debug)]
79enum SubCommand {
80    /// Apply migrations
81    Apply(Apply),
82    /// Drop migration information table. Needs all migrations to be
83    /// reverted else raises error
84    Drop,
85    /// List migrations along with their status and time applied if migrations
86    /// is already applied
87    List,
88    /// Revert migrations
89    Revert(Revert),
90}
91
92impl SubCommand {
93    async fn handle_subcommand<DB, M>(
94        &self,
95        migrator: &M,
96        connection: &mut <DB as Database>::Connection,
97    ) -> Result<(), Error>
98    where
99        DB: Database,
100        M: Migrate<DB> + ?Sized,
101    {
102        match self {
103            SubCommand::Apply(apply) => apply.run(connection, migrator).await?,
104            SubCommand::Drop => drop_migrations(connection, migrator).await?,
105            SubCommand::List => list_migrations(connection, migrator).await?,
106            SubCommand::Revert(revert) => revert.run(connection, migrator).await?,
107        }
108        Ok(())
109    }
110}

112async fn drop_migrations<DB, M>(
113    connection: &mut <DB as Database>::Connection,
114    migrator: &M,
115) -> Result<(), Error>
116where
117    DB: Database,
118    M: Migrate<DB> + ?Sized,
119{
120    migrator.ensure_migration_table_exists(connection).await?;
121    // hold lock for dropping migration table to avoid race condition where another
122    // process is applying migration and we are dropping the table
123    migrator.lock(connection).await?;
124    let result = async {
125        if !migrator
126            .fetch_applied_migration_from_db(connection)
127            .await?
128            .is_empty()
129        {
130            return Err(Error::AppliedMigrationExists);
131        }
132        migrator.drop_migration_table_if_exists(connection).await?;
133        Ok(())
134    }
135    .await;
136    let unlock_result = migrator.unlock(connection).await;
137    result.and(unlock_result)?;
138    println!("Dropped migrations table");
139    Ok(())
140}

142async fn list_migrations<DB, M>(
143    connection: &mut <DB as Database>::Connection,
144    migrator: &M,
145) -> Result<(), Error>
146where
147    DB: Database,
148    M: Migrate<DB> + ?Sized,
149{
150    migrator.ensure_migration_table_exists(connection).await?;
151    let applied_migrations = migrator.fetch_applied_migration_from_db(connection).await?;
152    let status_list = migrator.status(&applied_migrations)?;
153
154    let apply_plan_keys = if status_list.is_empty() {
155        HashSet::new()
156    } else {
157        migrator
158            .generate_migration_plan(Some(&Plan::apply_all()), &applied_migrations)?
159            .iter()
160            .map(|migration| (migration.app(), migration.name()))
161            .collect::<HashSet<_>>()
162    };
163
164    let widths = [5, 10, 50, 10, 40];
165    let full_width = widths.iter().sum::<usize>() + widths.len() * 3;
166
167    let first_width = widths[0];
168    let second_width = widths[1];
169    let third_width = widths[2];
170    let fourth_width = widths[3];
171    let fifth_width = widths[4];
172
173    println!(
174        "{:^first_width$} | {:^second_width$} | {:^third_width$} | {:^fourth_width$} | \
175         {:^fifth_width$}",
176        "ID", "App", "Name", "Status", "Applied time"
177    );
178
179    println!("{:^full_width$}", "-".repeat(full_width));
180    let known_keys = status_list
181        .iter()
182        .map(|(migration, _)| (migration.app(), migration.name()))
183        .collect::<HashSet<_>>();
184    for (migration, applied) in &status_list {
185        let mut id = String::from("N/A");
186        let mut status = "\u{2717}";
187        let mut applied_time = String::from("N/A");
188
189        if let Some(sqlx_migration) = applied {
190            id = sqlx_migration.id().to_string();
191            status = "\u{2713}";
192            applied_time = sqlx_migration.applied_time().to_string();
193        } else if !apply_plan_keys.contains(&(migration.app(), migration.name())) {
194            status = "\u{2194}";
195        }
196
197        println!(
198            "{:^first_width$} | {:^second_width$} | {:^third_width$} | {:^fourth_width$} | \
199             {:^fifth_width$}",
200            id,
201            migration.app(),
202            migration.name(),
203            status,
204            applied_time
205        );
206    }
207    // show applied migrations which no longer exist in the migration list
208    // (e.g. migration was removed from the codebase) so drift is visible.
209    // Such entries can be removed using apply --prune
210    for row in &applied_migrations {
211        if !known_keys.contains(&(row.app(), row.name())) {
212            println!(
213                "{:^first_width$} | {:^second_width$} | {:^third_width$} | {:^fourth_width$} | \
214                 {:^fifth_width$}",
215                row.id(),
216                row.app(),
217                row.name(),
218                "?",
219                row.applied_time()
220            );
221        }
222    }
223    Ok(())
224}

226#[derive(Parser, Debug)]
227#[expect(clippy::struct_excessive_bools)]
228struct Apply {
229    /// App name up to which migration needs to be applied. If migration option
230    /// is also present then only till migration is applied
231    #[arg(long)]
232    app: Option<String>,
233    /// Check for pending migration
234    #[arg(long, conflicts_with_all = ["count", "fake", "force", "plan"])]
235    check: bool,
236    /// Number of migration to apply. Conflicts with app args
237    #[arg(long, conflicts_with = "app")]
238    count: Option<usize>,
239    /// Make migration applied without running migration operations
240    #[arg(long)]
241    fake: bool,
242    /// Force run apply operation without asking question if migration is
243    /// destructible
244    #[arg(long)]
245    force: bool,
246    /// Apply migration till provided migration. Requires app options to be
247    /// present
248    #[arg(long, requires = "app")]
249    migration: Option<String>,
250    /// Show plan
251    #[arg(long, conflicts_with_all = ["fake", "force"])]
252    plan: bool,
253    /// Prune applied migration entries from the migration table which no
254    /// longer exist in the migration list after applying migrations
255    #[arg(long, conflicts_with_all = ["check", "plan"])]
256    prune: bool,
257}
258impl Apply {
259    async fn run<DB, M>(
260        &self,
261        connection: &mut <DB as Database>::Connection,
262        migrator: &M,
263    ) -> Result<(), Error>
264    where
265        DB: Database,
266        M: Migrate<DB> + ?Sized,
267    {
268        let plan = if let Some(count) = self.count {
269            Plan::apply_count(count)
270        } else if let Some(app) = &self.app {
271            Plan::apply_name(app, self.migration.as_deref())
272        } else {
273            Plan::apply_all()
274        }
275        .fake(self.fake);
276        migrator.ensure_migration_table_exists(connection).await?;
277        let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
278        let migrations = migrator.generate_migration_plan(Some(&plan), &applied_rows)?;
279        if self.check {
280            if !migrations.is_empty() {
281                return Err(Error::PendingMigrationPresent);
282            }
283            println!("No pending migration to apply");
284            return Ok(());
285        }
286        if self.plan {
287            if migrations.is_empty() {
288                println!("No migration exists for applying");
289            } else {
290                let first_width = 10;
291                let second_width = 50;
292                let full_width = first_width + second_width + 3;
293                println!("{:^first_width$} | {:^second_width$}", "App", "Name");
294                println!("{:^full_width$}", "-".repeat(full_width));
295                for migration in migrations {
296                    println!(
297                        "{:^first_width$} | {:^second_width$}",
298                        migration.app(),
299                        migration.name(),
300                    );
301                }
302            }
303        } else {
304            let destructible_migrations = migrations
305                .iter()
306                .filter(|m| m.operations().iter().any(|o| o.is_destructible()))
307                .collect::<Vec<_>>();
308            if !self.force && !destructible_migrations.is_empty() && !self.fake {
309                if !std::io::stdin().is_terminal() {
310                    return Err(Error::ConfirmationRequired);
311                }
312                let mut input = String::new();
313                println!(
314                    "Do you want to apply {} destructible migrations? (y/N)",
315                    destructible_migrations.len()
316                );
317                for (position, migration) in destructible_migrations.iter().enumerate() {
318                    println!(
319                        "{}. {} : {}",
320                        position + 1,
321                        migration.app(),
322                        migration.name()
323                    );
324                }
325                std::io::stdout().flush()?;
326                std::io::stdin().read_line(&mut input)?;
327                let input_trimmed = input.trim().to_ascii_lowercase();
328                // If answer is not y or yes then return
329                if !["y", "yes"].contains(&input_trimmed.as_str()) {
330                    println!("Aborted applying migrations");
331                    return Ok(());
332                }
333            }
334            migrator.run(connection, &plan).await?;
335            println!("Successfully applied migrations according to plan");
336            if self.prune {
337                let pruned_count = migrator.prune(connection).await?;
338                println!("Pruned {pruned_count} unknown migrations from migration table");
339            }
340        }
341        Ok(())
342    }
343}

345#[derive(Parser, Debug)]
346#[expect(clippy::struct_excessive_bools)]
347struct Revert {
348    /// Revert all migration. Conflicts with app args
349    #[arg(long, conflicts_with = "app")]
350    all: bool,
351    /// Revert migration till app migrations is reverted. If it is present
352    /// alongside migration options then only till migration is reverted
353    #[arg(long)]
354    app: Option<String>,
355    /// Number of migration to revert. Conflicts with all and app args
356    #[arg(long, conflicts_with_all = ["all", "app"])]
357    count: Option<usize>,
358    /// Make migration reverted without running revert operation
359    #[arg(long)]
360    fake: bool,
361    /// Force run revert operation without asking question
362    #[arg(long)]
363    force: bool,
364    /// Revert migration till provided migration. Requires app options to be
365    /// present
366    #[arg(long, requires = "app")]
367    migration: Option<String>,
368    /// Show plan
369    #[arg(long, conflicts_with_all = ["fake", "force"])]
370    plan: bool,
371}
372impl Revert {
373    async fn run<DB, M>(
374        &self,
375        connection: &mut <DB as Database>::Connection,
376        migrator: &M,
377    ) -> Result<(), Error>
378    where
379        DB: Database,
380        M: Migrate<DB> + ?Sized,
381    {
382        let plan = if let Some(count) = self.count {
383            Plan::revert_count(count)
384        } else if let Some(app) = &self.app {
385            Plan::revert_name(app, self.migration.as_deref())
386        } else if self.all {
387            Plan::revert_all()
388        } else {
389            Plan::revert_count(1)
390        }
391        .fake(self.fake);
392        migrator.ensure_migration_table_exists(connection).await?;
393        let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
394        let revert_migrations = migrator.generate_migration_plan(Some(&plan), &applied_rows)?;
395
396        if self.plan {
397            if revert_migrations.is_empty() {
398                println!("No migration exists for reverting");
399            } else {
400                let first_width = 10;
401                let second_width = 50;
402                let full_width = first_width + second_width + 3;
403                println!("{:^first_width$} | {:^second_width$}", "App", "Name");
404                println!("{:^full_width$}", "-".repeat(full_width));
405                for migration in revert_migrations {
406                    println!(
407                        "{:^first_width$} | {:^second_width$}",
408                        migration.app(),
409                        migration.name(),
410                    );
411                }
412            }
413        } else {
414            if !self.force && !revert_migrations.is_empty() && !self.fake {
415                if !std::io::stdin().is_terminal() {
416                    return Err(Error::ConfirmationRequired);
417                }
418                let mut input = String::new();
419                println!(
420                    "Do you want to revert {} migrations? (y/N)",
421                    revert_migrations.len()
422                );
423                for (position, migration) in revert_migrations.iter().enumerate() {
424                    println!(
425                        "{}. {} : {}",
426                        position + 1,
427                        migration.app(),
428                        migration.name()
429                    );
430                }
431                std::io::stdout().flush()?;
432                std::io::stdin().read_line(&mut input)?;
433                let input_trimmed = input.trim().to_ascii_lowercase();
434                // If answer is not y or yes then return
435                if !["y", "yes"].contains(&input_trimmed.as_str()) {
436                    println!("Aborted reverting migrations");
437                    return Ok(());
438                }
439            }
440            migrator.run(connection, &plan).await?;
441            println!("Successfully reverted migrations according to plan");
442        }
443        Ok(())
444    }
