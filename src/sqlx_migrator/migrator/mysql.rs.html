7/// Internal migrations in order, see [`super::internal_migration`]
8fn internal_migration_steps(table_name: &str) -> Vec<(&'static str, String)> {
9    vec![(
10        "0001_create_migrations_table",
11        create_migrations_table(table_name),
12    )]
13}
14
15fn create_migrations_table(table_name: &str) -> String {
16    format!(
17        "CREATE TABLE IF NOT EXISTS {table_name} (
18        id INT PRIMARY KEY NOT NULL AUTO_INCREMENT,
19        app VARCHAR(384) NOT NULL,
20        name VARCHAR(384) NOT NULL,
21        applied_time TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
22        UNIQUE (app, name)
23    )"
24    )
25}

27/// Internal migrations for the given table names
28fn internal_migrations(table_name: &str, meta_table_name: &str) -> InternalMigrations {
29    let create_meta_table = format!(
30        "CREATE TABLE IF NOT EXISTS {meta_table_name} (name VARCHAR(384) PRIMARY KEY NOT NULL, \
31         applied_time TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP)"
32    );
33    let fetch_applied = format!("SELECT name FROM {meta_table_name}");
34    let insert_applied = format!("INSERT IGNORE INTO {meta_table_name}(name) VALUES (?)");
35    InternalMigrations::new(
36        create_meta_table,
37        fetch_applied,
38        insert_applied,
39        internal_migration_steps(table_name),
40    )
41}

43/// Drop table query
44#[must_use]
45fn drop_table_query(table_name: &str) -> String {
46    format!("DROP TABLE IF EXISTS {table_name}")
47}
48
49/// Fetch rows query
50#[must_use]
51fn fetch_rows_query(table_name: &str) -> String {
52    format!(
53        "SELECT id, app, name, DATE_FORMAT(applied_time, '%Y-%m-%d %H:%i:%s') AS applied_time \
54         FROM {table_name}"
55    )
56}
57
58/// Add migration query
59#[must_use]
60fn add_migration_query(table_name: &str) -> String {
61    format!("INSERT INTO {table_name}(app, name) VALUES (?, ?)")
62}
63
64/// Delete migration query
65#[must_use]
66fn delete_migration_query(table_name: &str) -> String {
67    format!("DELETE FROM {table_name} WHERE app = ? AND name = ?")
68}

70/// Built in SQL, see [`Dialect::mysql`]
71fn sql(names: &Names) -> Sql {
72    // lock names are server wide and at most 64 characters, so the database
73    // name is included and the whole key is hashed
74    let lock = Lock::new(
75        "SELECT GET_LOCK(CRC32(CONCAT(COALESCE(DATABASE(), ''), '/', ?)), -1)",
76        "SELECT RELEASE_LOCK(CRC32(CONCAT(COALESCE(DATABASE(), ''), '/', ?)))",
77    );
78    Sql {
79        create_schema: None,
80        internal: internal_migrations(&names.table(), &names.meta_table()),
81        drop_tables: format!(
82            "{}; {}",
83            drop_table_query(&names.table()),
84            drop_table_query(&names.meta_table())
85        ),
86        fetch_rows: fetch_rows_query(&names.table()),
87        add_migration: add_migration_query(&names.table()),
88        delete_migration: delete_migration_query(&names.table()),
89        lock: Some(lock),
90    }
91}

93/// Built in dialect, see [`Dialect::mysql`]
94pub(crate) fn dialect(names: Names) -> Dialect<MySql> {
95    Dialect::new(names, sql)
96}
