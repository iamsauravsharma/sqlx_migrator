13/// SQL of one backend with table names filled in
14#[derive(Debug, Clone)]
15pub(crate) struct Sql {
16    /// Creates the schema if missing, `None` without a schema or on backends
17    /// which cannot create one
18    pub(crate) create_schema: Option<String>,
19    /// Internal migrations which create and change the migrator tables
20    pub(crate) internal: InternalMigrations,
21    /// Drops the migration and meta tables if they exist
22    pub(crate) drop_tables: String,
23    /// Selects the applied migration rows as `id, app, name, applied_time`
24    pub(crate) fetch_rows: String,
25    /// Records an applied migration. Binds app then name
26    pub(crate) add_migration: String,
27    /// Removes an applied migration. Binds app then name
28    pub(crate) delete_migration: String,
29    /// Statements of the advisory lock, `None` on backends without one
30    pub(crate) lock: Option<Lock>,
31}

33const DEFAULT_TABLE_NAME: &str = "_sqlx_migrator_migrations";
34const DEFAULT_META_TABLE_NAME: &str = "_sqlx_migrator_meta";

36/// Validated table prefix and schema the SQL of a backend is built for
37#[derive(Debug, Clone, Default)]
38pub(crate) struct Names {
39    /// Table prefix
40    table_prefix: Option<String>,
41    /// Schema the tables live in
42    schema: Option<String>,
43}
44
45impl Names {
46    /// Names for an optional table prefix and schema
47    ///
48    /// # Errors
49    /// When the table prefix or the schema contains invalid characters
50    pub(crate) fn new(table_prefix: Option<&str>, schema: Option<&str>) -> Result<Self, Error> {
51        let is_name_char = |c: char| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_';
52        if table_prefix.is_some_and(|prefix| prefix.is_empty() || !prefix.chars().all(is_name_char))
53        {
54            return Err(Error::InvalidTablePrefix);
55        }
56        if schema.is_some_and(|schema| {
57            !schema.starts_with(|c: char| c.is_ascii_lowercase() || c == '_')
58                || !schema.chars().all(is_name_char)
59        }) {
60            return Err(Error::InvalidSchema);
61        }
62        Ok(Self {
63            table_prefix: table_prefix.map(str::to_string),
64            schema: schema.map(str::to_string),
65        })
66    }
67
68    /// Table prefix, if any
69    pub(crate) fn table_prefix(&self) -> Option<&str> {
70        self.table_prefix.as_deref()
71    }
72
73    /// Schema the tables live in, if any
74    pub(crate) fn schema(&self) -> Option<&str> {
75        self.schema.as_deref()
76    }
77
78    /// Applies the prefix and schema to a base table name
79    fn qualified(&self, base: &str) -> String {
80        let name = self
81            .table_prefix
82            .as_ref()
83            .map_or(base.to_string(), |prefix| format!("_{prefix}{base}"));
84        self.schema
85            .as_ref()
86            .map_or(name.clone(), |schema| format!("{schema}.{name}"))
87    }
88
89    /// Name of the migration table
90    pub(crate) fn table(&self) -> String {
91        self.qualified(DEFAULT_TABLE_NAME)
92    }
93
94    /// Name of the meta table
95    pub(crate) fn meta_table(&self) -> String {
96        self.qualified(DEFAULT_META_TABLE_NAME)
97    }
98}

100/// Advisory lock held while a plan runs.
101#[derive(Debug, Clone)]
102pub(crate) struct Lock {
103    acquire: String,
104    release: String,
105}
106
107impl Lock {
108    /// Lock from a statement which takes it, waiting until it is free, and
109    /// one which releases it
110    #[cfg(any(feature = "postgres", feature = "mysql"))]
111    pub(crate) fn new(acquire: impl Into<String>, release: impl Into<String>) -> Self {
112        Self {
113            acquire: acquire.into(),
114            release: release.into(),
115        }
116    }
117
118    /// Statement which takes the lock
119    pub(crate) fn acquire(&self) -> &str {
120        &self.acquire
121    }
122
123    /// Statement which releases the lock
124    pub(crate) fn release(&self) -> &str {
125        &self.release
126    }
127}

129/// Configuration of the migrator for database `DB`: table names, advisory
130/// lock and DDL behaviour, together with the SQL built for them.
131///
132/// Use [`Dialect::postgres`], [`Dialect::mysql`] or [`Dialect::sqlite`] for the
133/// built in configuration, or the matching `_with` constructor to also set a
134/// table prefix and schema. Those are fixed once the dialect is built. Options
135/// which only some databases have, such as the advisory lock, can only be
136/// set on the dialect of a database which has them.
137pub struct Dialect<DB> {
138    names: Names,
139    sql: Sql,
140    lock: bool,
141    create_schema: bool,
142    database: PhantomData<DB>,
143}
144
145impl<DB> Clone for Dialect<DB> {
146    fn clone(&self) -> Self {
147        Self {
148            names: self.names.clone(),
149            sql: self.sql.clone(),
150            lock: self.lock,
151            create_schema: self.create_schema,
152            database: PhantomData,
153        }
154    }
155}
156
157impl<DB> fmt::Debug for Dialect<DB> {
158    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
159        f.debug_struct("Dialect")
160            .field("names", &self.names)
161            .field("sql", &self.sql)
162            .field("lock", &self.lock)
163            .field("create_schema", &self.create_schema)
164            .finish()
165    }
166}

168impl<DB> Dialect<DB> {
169    /// Dialect for `names` whose SQL is built once by `build_sql`
170    pub(crate) fn new(names: Names, build_sql: fn(&Names) -> Sql) -> Self {
171        let sql = build_sql(&names);
172        Self {
173            names,
174            sql,
175            lock: true,
176            create_schema: true,
177            database: PhantomData,
178        }
179    }
180
181    /// The SQL run by the migrator
182    pub(crate) fn sql(&self) -> &Sql {
183        &self.sql
184    }
185
186    /// Lock statements when the lock is enabled and the backend has one
187    pub(crate) fn lock_statements(&self) -> Option<&Lock> {
188        self.sql.lock.as_ref().filter(|_| self.lock)
189    }
190
191    /// Statement creating the schema when enabled and the backend has one
192    pub(crate) fn create_schema_statement(&self) -> Option<&str> {
193        self.sql
194            .create_schema
195            .as_deref()
196            .filter(|_| self.create_schema)
197    }
198
199    /// Name of the table which stores applied migrations
200    ///
201    /// Format depends on the prefix and schema the dialect was built with:
202    /// - With schema: `{schema}._sqlx_migrator_migrations`
203    /// - With prefix: `_{prefix}_sqlx_migrator_migrations`
204    /// - With both: `{schema}._{prefix}_sqlx_migrator_migrations`
205    /// - Default: `_sqlx_migrator_migrations`
206    #[must_use]
207    pub fn table_name(&self) -> String {
208        self.names.table()
209    }
210
211    /// Name of the meta table which records the internal migrations. It
212    /// follows the same prefix and schema rules as [`Dialect::table_name`].
213    #[must_use]
214    pub fn meta_table_name(&self) -> String {
215        self.names.meta_table()
216    }
217
218    /// Name of the schema the tables live in, if any
219    #[must_use]
220    pub fn schema(&self) -> Option<&str> {
221        self.names.schema()
222    }
223
224    /// Table prefix the table names were built with, if any
225    #[must_use]
226    pub fn table_prefix(&self) -> Option<&str> {
227        self.names.table_prefix()
228    }
229}

231/// Advisory lock option of databases which have advisory locks
232#[cfg(any(feature = "postgres", feature = "mysql"))]
233macro_rules! lock_option {
234    ($database:ty) => {
235        impl Dialect<$database> {
236            /// Whether the advisory lock is held while a plan runs
237            #[must_use]
238            pub fn lock(&self) -> bool {
239                self.lock
240            }
241
242            /// Sets whether the advisory lock is held while a plan runs. Use
243            /// `false` behind a transaction pooling proxy such as `PgBouncer`
244            #[must_use]
245            pub fn set_lock(mut self, lock: bool) -> Self {
246                self.lock = lock;
247                self
248            }
249        }
250    };
251}

273        Names::new(table_prefix, schema).map(super::postgres::dialect)
274    }
275
276    /// Whether the schema is created if missing before the tables are
277    /// ensured
278    #[must_use]
279    pub fn create_schema(&self) -> bool {
280        self.create_schema
281    }
282
283    /// Sets whether the schema is created if missing. Use `false` when the
284    /// schema is managed elsewhere, for example when the database user is not
285    /// allowed to create schemas
286    #[must_use]
287    pub fn set_create_schema(mut self, create_schema: bool) -> Self {
288        self.create_schema = create_schema;
289        self
290    }

330/// A sqlx database the migrator can execute a [`Dialect`] against.
331///
332/// Implemented for every database whose connection runs plain and bound
333/// queries, which covers Postgres, MySQL and `SQLite`.
334#[async_trait::async_trait]
335pub trait SupportedDatabase: Database + Send + Sync {
336    /// Runs one or more statements without parameters
337    async fn execute(connection: &mut Self::Connection, sql: &str) -> Result<(), Error>;
338
339    /// Runs one statement binding the given text parameters in order
340    async fn execute_bound(
341        connection: &mut Self::Connection,
342        sql: &str,
343        binds: &[&str],
344    ) -> Result<(), Error>;
345
346    /// Fetches the first column of every row as text
347    async fn fetch_text(connection: &mut Self::Connection, sql: &str)
348    -> Result<Vec<String>, Error>;
349
350    /// Fetches applied migration rows
351    async fn fetch_rows(
352        connection: &mut Self::Connection,
353        sql: &str,
354    ) -> Result<Vec<AppliedMigrationSqlRow>, Error>;
355}
356
357#[async_trait::async_trait]
358impl<DB> SupportedDatabase for DB
359where
360    DB: Database + Send + Sync,
361    DB::Arguments: IntoArguments<DB>,
362    for<'c> &'c mut DB::Connection: Executor<'c, Database = DB>,
363    for<'q> String: sqlx::Encode<'q, DB> + sqlx::Type<DB>,
364    for<'r> (String,): sqlx::FromRow<'r, DB::Row>,
365    for<'r> AppliedMigrationSqlRow: sqlx::FromRow<'r, DB::Row>,
366{
367    async fn execute(connection: &mut Self::Connection, sql: &str) -> Result<(), Error> {
368        sqlx::raw_sql(AssertSqlSafe(sql.to_string()))
369            .execute(connection)
370            .await?;
371        Ok(())
372    }
373
374    async fn execute_bound(
375        connection: &mut Self::Connection,
376        sql: &str,
377        binds: &[&str],
378    ) -> Result<(), Error> {
379        let mut query = sqlx::query(AssertSqlSafe(sql.to_string()));
380        for bind in binds {
381            query = query.bind((*bind).to_string());
382        }
383        query.execute(connection).await?;
384        Ok(())
385    }
386
387    async fn fetch_text(
388        connection: &mut Self::Connection,
389        sql: &str,
390    ) -> Result<Vec<String>, Error> {
391        let rows: Vec<(String,)> = sqlx::query_as(AssertSqlSafe(sql.to_string()))
392            .fetch_all(connection)
393            .await?;
394        Ok(rows.into_iter().map(|(value,)| value).collect())
395    }
396
397    async fn fetch_rows(
398        connection: &mut Self::Connection,
399        sql: &str,
400    ) -> Result<Vec<AppliedMigrationSqlRow>, Error> {
401        Ok(sqlx::query_as(AssertSqlSafe(sql.to_string()))
402            .fetch_all(connection)
403            .await?)
404    }
