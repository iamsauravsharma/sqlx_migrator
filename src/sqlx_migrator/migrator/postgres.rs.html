7/// Create schema query
8#[must_use]
9fn create_schema_query(schema_name: &str) -> String {
10    format!("CREATE SCHEMA IF NOT EXISTS {schema_name}")
11}

13/// Internal migrations in order, see [`super::internal_migration`]
14fn internal_migration_steps(table_name: &str) -> Vec<(&'static str, String)> {
15    vec![
16        (
17            "0001_create_migrations_table",
18            create_migrations_table(table_name),
19        ),
20        (
21            "0002_add_success_column",
22            format!(
23                "ALTER TABLE {table_name} ADD COLUMN IF NOT EXISTS success BOOLEAN NOT NULL \
24                 DEFAULT TRUE"
25            ),
26        ),
27        (
28            "0003_add_execution_time_column",
29            format!(
30                "ALTER TABLE {table_name} ADD COLUMN IF NOT EXISTS execution_time_ms BIGINT NOT \
31                 NULL DEFAULT 0"
32            ),
33        ),
34        (
35            "0004_add_batch_column",
36            format!(
37                "ALTER TABLE {table_name} ADD COLUMN IF NOT EXISTS batch INT NOT NULL DEFAULT 0; \
38                 UPDATE {table_name} SET batch = id"
39            ),
40        ),
41    ]
42}

44fn create_migrations_table(table_name: &str) -> String {
45    format!(
46        "CREATE TABLE IF NOT EXISTS {table_name} (
47        id INT PRIMARY KEY NOT NULL GENERATED ALWAYS AS IDENTITY,
48        app TEXT NOT NULL,
49        name TEXT NOT NULL,
50        applied_time TIMESTAMPTZ NOT NULL DEFAULT now(),
51        UNIQUE (app, name)
52    )"
53    )
54}

56/// Internal migrations for the given table names
57fn internal_migrations(table_name: &str, meta_table_name: &str) -> InternalMigrations {
58    let create_meta_table = format!(
59        "CREATE TABLE IF NOT EXISTS {meta_table_name} (name TEXT PRIMARY KEY NOT NULL, \
60         applied_time TIMESTAMPTZ NOT NULL DEFAULT now())"
61    );
62    let fetch_applied = format!("SELECT name FROM {meta_table_name}");
63    let insert_applied =
64        format!("INSERT INTO {meta_table_name}(name) VALUES ($1) ON CONFLICT (name) DO NOTHING");
65    InternalMigrations::new(
66        create_meta_table,
67        fetch_applied,
68        insert_applied,
69        internal_migration_steps(table_name),
70    )
71}

73/// Drop table query
74#[must_use]
75fn drop_table_query(table_name: &str) -> String {
76    format!("DROP TABLE IF EXISTS {table_name}")
77}

79/// Fetch rows query
80#[must_use]
81fn fetch_rows_query(table_name: &str) -> String {
82    format!(
83        "SELECT id, app, name, applied_time::TEXT, success, execution_time_ms, batch FROM \
84         {table_name}"
85    )
86}

88/// Add migration query
89#[must_use]
90fn add_migration_query(table_name: &str) -> String {
91    format!(
92        "INSERT INTO {table_name}(app, name, execution_time_ms, batch) VALUES ($1, $2, CAST($3 AS \
93         BIGINT), CAST($4 AS INT))"
94    )
95}

97/// Start migration query, records a migration which is running as failed
98#[must_use]
99fn start_migration_query(table_name: &str) -> String {
100    format!(
101        "INSERT INTO {table_name}(app, name, success, batch) VALUES ($1, $2, FALSE, CAST($3 AS \
102         INT))"
103    )
104}
105
106/// Finish migration query, marks a running migration as successful
107#[must_use]
108fn finish_migration_query(table_name: &str) -> String {
109    format!(
110        "UPDATE {table_name} SET success = TRUE, execution_time_ms = CAST($1 AS BIGINT) WHERE app \
111         = $2 AND name = $3"
112    )
113}

115/// Delete migration query
116#[must_use]
117fn delete_migration_query(table_name: &str) -> String {
118    format!("DELETE FROM {table_name} WHERE app = $1 AND name = $2")
119}

121/// Delete failed migrations query
122#[must_use]
123fn delete_failed_migrations_query(table_name: &str) -> String {
124    format!("DELETE FROM {table_name} WHERE success = FALSE")
125}

127/// Row lock statements for the given lock table name
128fn row_lock(lock_table: &str) -> RowLock {
129    RowLock {
130        create_table: format!(
131            "CREATE TABLE IF NOT EXISTS {lock_table} (name TEXT PRIMARY KEY NOT NULL, locked_at \
132             TIMESTAMPTZ NOT NULL DEFAULT now())"
133        ),
134        acquire: format!("INSERT INTO {lock_table}(name) VALUES ($1)"),
135        release: format!("DELETE FROM {lock_table} WHERE name = $1"),
136    }
137}

139/// Lock table query, holds the lock until the transaction ends
140#[must_use]
141fn lock_table_query(table_name: &str, mode: &str) -> String {
142    format!("LOCK TABLE {table_name} IN {mode} MODE")
143}

145/// Built in SQL, see [`Dialect::postgres`]
146fn sql(names: &Names) -> Sql {
147    // advisory locks are scoped to the connected database already
148    let lock = Lock::new(
149        "SELECT pg_advisory_lock(hashtext($1))",
150        "SELECT pg_advisory_unlock(hashtext($1))",
151    );
152    Sql {
153        create_schema: names.schema().map(create_schema_query),
154        internal: internal_migrations(&names.table(), &names.meta_table()),
155        drop_tables: format!(
156            "{}; {}",
157            drop_table_query(&names.table()),
158            drop_table_query(&names.meta_table())
159        ),
160        fetch_rows: fetch_rows_query(&names.table()),
161        add_migration: add_migration_query(&names.table()),
162        start_migration: start_migration_query(&names.table()),
163        finish_migration: finish_migration_query(&names.table()),
164        delete_migration: delete_migration_query(&names.table()),
165        delete_failed_migrations: delete_failed_migrations_query(&names.table()),
166        lock: Some(lock),
167        row_lock: row_lock(&names.lock_table()),
168        transaction_locks: Some(TransactionLocks {
169            advisory: "SELECT pg_advisory_xact_lock(hashtext($1))".to_string(),
170            // reading a catalog table makes the session take in what was
171            // committed while it waited
172            after_lock: "SELECT 1 FROM pg_catalog.pg_namespace LIMIT 1".to_string(),
173            share_update_exclusive: lock_table_query(&names.table(), "SHARE UPDATE EXCLUSIVE"),
174            access_exclusive: lock_table_query(&names.table(), "ACCESS EXCLUSIVE"),
175        }),
176        drop_lock_table: drop_table_query(&names.lock_table()),
177    }
178}

180/// Built in dialect, see [`Dialect::postgres`]
181pub(crate) fn dialect(names: Names) -> Dialect<Postgres> {
182    Dialect::new(names, sql)
183}
