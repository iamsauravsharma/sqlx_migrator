80/// Struct that determines the type of migration plan to execute.
81///
82/// A [`Plan`] can specify whether to apply or revert migrations, and may target
83/// all migrations, specific migrations, or a limited number of migrations.
84#[derive(Debug)]
85pub struct Plan {
86    #[expect(
87        clippy::struct_field_names,
88        reason = "type is a keyword so it cannot be used"
89    )]
90    plan_type: PlanType,
91    app_migration: Option<(String, Option<String>)>,
92    count: Option<usize>,
93    fake: bool,
94}
95
96impl Plan {
97    fn new(
98        plan_type: PlanType,
99        app_migration: Option<(String, Option<String>)>,
100        count: Option<usize>,
101    ) -> Self {
102        Self {
103            plan_type,
104            app_migration,
105            count,
106            fake: false,
107        }
108    }
109
110    /// Sets the plan as a "fake" plan.
111    ///
112    /// When the plan is marked as fake, the migration status is updated to
113    /// either "applied" or "reverted" without actually performing any
114    /// migration operations. This is useful for scenarios where you want to
115    /// simulate the effect of applying or reverting a migration, but
116    /// without making changes to the database.
117    ///
118    /// By default, the `fake` flag is set to `false`, and the migration
119    /// operations are executed as expected.
120    #[must_use]
121    pub fn fake(mut self, fake: bool) -> Self {
122        self.fake = fake;
123        self
124    }
125
126    /// Creates a new plan to apply all migrations.
127    #[must_use]
128    pub fn apply_all() -> Self {
129        Self::new(PlanType::Apply, None, None)
130    }
131
132    /// Creates a new plan to apply a specific migration by name. If migration
133    /// name is not provided it will apply all migrations of the app
134    #[must_use]
135    pub fn apply_name(app: &str, name: Option<&str>) -> Self {
136        Self::new(
137            PlanType::Apply,
138            Some((app.to_string(), name.map(ToString::to_string))),
139            None,
140        )
141    }
142
143    /// Creates a new plan to apply a limited number of migrations.
144    #[must_use]
145    pub fn apply_count(count: usize) -> Self {
146        Self::new(PlanType::Apply, None, Some(count))
147    }
148
149    /// Creates a new plan to revert all migrations.
150    #[must_use]
151    pub fn revert_all() -> Self {
152        Self::new(PlanType::Revert(RevertPlan::default()), None, None)
153    }
154
155    /// Creates a new plan to revert a specific migration by name. If migration
156    /// name is not provided it will revert all migrations of the app
157    #[must_use]
158    pub fn revert_name(app: &str, name: Option<&str>) -> Self {
159        Self::new(
160            PlanType::Revert(RevertPlan::default()),
161            Some((app.to_string(), name.map(ToString::to_string))),
162            None,
163        )
164    }
165
166    /// Creates a new plan to revert a limited number of migrations.
167    #[must_use]
168    pub fn revert_count(count: usize) -> Self {
169        Self::new(PlanType::Revert(RevertPlan::default()), None, Some(count))
170    }
171
172    /// Creates a new plan to revert the migrations applied by the last run,
173    /// which share the highest batch number
174    #[must_use]
175    pub fn revert_last_batch() -> Self {
176        let plan_type = PlanType::Revert(RevertPlan { last_batch: true });
177        Self::new(plan_type, None, None)
178    }
179}

259/// What a lock is held on
260enum Held<'connection, DB: Database> {
261    /// The connection, the lock is released by statements or nothing was taken
262    Connection(&'connection mut <DB as Database>::Connection),
263    /// A transaction which holds the lock until it ends
264    Transaction(sqlx::Transaction<'connection, DB>),
265}
266
267/// A lock taken by [`Migrator::lock`]
268pub(crate) struct Locked<'connection, DB: Database>(Held<'connection, DB>);
269
270impl<DB: Database> Locked<'_, DB> {
271    /// Connection to run what needs the lock on
272    pub(crate) fn connection(&mut self) -> &mut <DB as Database>::Connection {
273        match &mut self.0 {
274            Held::Connection(connection) => connection,
275            Held::Transaction(transaction) => transaction,
276        }
277    }
278}

329    /// Adds a list of migrations to the migrator.
330    ///
331    /// This method accepts a vector of migrations and adds each one
332    /// individually to ensure proper handling of migration relationships
333    /// and duplicates.
334    ///
335    /// # Errors
336    /// If migration is added with same app and name but inconsistent value i.e
337    /// its parents, run before, replaces and is atomic differ and do not have
338    /// same number of operation
339    pub fn add_migrations(&mut self, migrations: Vec<BoxMigration<DB>>) -> Result<(), Error> {
340        for migration in migrations {
341            self.add_migration(migration)?;
342        }
343        Ok(())
344    }

346    /// Adds a single migration to the migrator.
347    ///
348    /// # Errors
349    /// If migration is added with same app and name but inconsistent value i.e
350    /// its parents, run before, replaces and is atomic differ and do not have
351    /// same number of operation
352    pub fn add_migration(&mut self, migration: BoxMigration<DB>) -> Result<(), Error> {
353        if migration.is_virtual() {
354            // Virtual migrations must not have any fields other than app/name
355            if !migration.parents().is_empty()
356                || !migration.operations().is_empty()
357                || !migration.replaces().is_empty()
358                || !migration.run_before().is_empty()
359            {
360                return Err(Error::InvalidVirtualMigration);
361            }
362            // Only add virtual migration if not already present
363            if self.migrations.contains(&migration) {
364                return Ok(());
365            }
366            self.migrations.push(migration);
367        } else {
368            let mut skip_add = false;
369            if let Some((migration_index, found_migration)) = self
370                .migrations()
371                .iter()
372                .enumerate()
373                .find(|(_, elem)| elem == &&migration)
374            {
375                if found_migration.is_virtual() {
376                    // Replace the placeholder virtual migration with the
377                    // concrete one
378                    self.migrations.remove(migration_index);
379                } else if found_migration.parents() != migration.parents()
380                    || found_migration.operations().len() != migration.operations().len()
381                    || found_migration.replaces() != migration.replaces()
382                    || found_migration.run_before() != migration.run_before()
383                    || found_migration.is_atomic() != migration.is_atomic()
384                {
385                    // Non-virtual duplicate with different definition
386                    return Err(Error::InconsistentMigration {
387                        app: migration.app().to_string(),
388                        name: migration.name().to_string(),
389                    });
390                } else {
391                    // Non-virtual consistent duplicate — already registered
392                    skip_add = true;
393                }
394            }
395
396            if !skip_add {
397                // Add this migration and recursively register its parents,
398                // replaces, and run_before migrations
399                let migration_parents = migration.parents();
400                let migration_replaces = migration.replaces();
401                let migration_run_before = migration.run_before();
402
403                self.migrations.push(migration);
404
405                for parent in migration_parents {
406                    self.add_migration(parent)?;
407                }
408                for replace in migration_replaces {
409                    self.add_migration(replace)?;
410                }
411                for run_before in migration_run_before {
412                    self.add_migration(run_before)?;
413                }
414            }
415        }
416        Ok(())
417    }

600    /// Take the lock of the dialect's [`LockStrategy`]
601    ///
602    /// # Errors
603    /// If the SQL fails or the lock row is held by another run
604    pub(crate) async fn lock<'connection>(
605        &self,
606        connection: &'connection mut <DB as Database>::Connection,
607    ) -> Result<Locked<'connection, DB>, Error> {
608        match self.dialect.lock_strategy() {
609            LockStrategy::None => {}
610            LockStrategy::Advisory => {
611                if let Some(lock) = self.dialect.advisory_lock() {
612                    DB::execute_bound(connection, lock.acquire(), &[&self.table_name()]).await?;
613                }
614            }
615            LockStrategy::Row => self.take_row_lock(connection).await?,
616            strategy @ (LockStrategy::TransactionAdvisory
617            | LockStrategy::TableShareUpdateExclusive
618            | LockStrategy::TableAccessExclusive) => {
619                return self.begin_locked_transaction(connection, strategy).await;
620            }
621        }
622        Ok(Locked(Held::Connection(connection)))
623    }

625    /// Release the lock taken by [`Migrator::lock`]. A transaction lock commits
626    /// its transaction when `succeeded` and rolls it back otherwise
627    ///
628    /// # Errors
629    /// If the SQL fails or the transaction cannot be committed
630    pub(crate) async fn unlock(
631        &self,
632        locked: Locked<'_, DB>,
633        succeeded: bool,
634    ) -> Result<(), Error> {
635        match locked.0 {
636            Held::Transaction(transaction) => {
637                if succeeded {
638                    transaction.commit().await?;
639                } else {
640                    transaction.rollback().await?;
641                }
642            }
643            Held::Connection(connection) => {
644                match self.dialect.lock_strategy() {
645                    LockStrategy::Advisory => {
646                        if let Some(lock) = self.dialect.advisory_lock() {
647                            DB::execute_bound(connection, lock.release(), &[&self.table_name()])
648                                .await?;
649                        }
650                    }
651                    LockStrategy::Row => {
652                        DB::execute_bound(
653                            connection,
654                            &self.dialect.sql().row_lock.release,
655                            &[&self.table_name()],
656                        )
657                        .await?;
658                    }
659                    _ => {}
660                }
661            }
662        }
663        Ok(())
664    }

666    /// Begin the transaction holding the lock of a transaction lock strategy
667    async fn begin_locked_transaction<'connection>(
668        &self,
669        connection: &'connection mut <DB as Database>::Connection,
670        strategy: LockStrategy,
671    ) -> Result<Locked<'connection, DB>, Error> {
672        let Some(locks) = &self.dialect.sql().transaction_locks else {
673            return Err(Error::UnsupportedLockStrategy { strategy });
674        };
675        let mut transaction = connection.begin().await?;
676        // the advisory lock keeps two nodes from creating the tables at once,
677        // which a table lock cannot since the table has to exist to be locked
678        DB::execute_bound(&mut transaction, &locks.advisory, &[&self.table_name()]).await?;
679        DB::execute(&mut transaction, &locks.after_lock).await?;
680        let table_lock = match strategy {
681            LockStrategy::TableShareUpdateExclusive => Some(&locks.share_update_exclusive),
682            LockStrategy::TableAccessExclusive => Some(&locks.access_exclusive),
683            _ => None,
684        };
685        if let Some(table_lock) = table_lock {
686            self.ensure_migration_table_exists(&mut transaction).await?;
687            DB::execute(&mut transaction, table_lock).await?;
688        }
689        Ok(Locked(Held::Transaction(transaction)))
690    }

692    /// Insert the lock row
693    async fn take_row_lock(
694        &self,
695        connection: &mut <DB as Database>::Connection,
696    ) -> Result<(), Error> {
697        let sql = self.dialect.sql();
698        if let Some(create_schema) = self.dialect.create_schema_statement() {
699            DB::execute(connection, create_schema).await?;
700        }
701        // two processes creating the table at once may fail one of them, the
702        // statement taking the lock then reports a table which is missing
703        if let Err(error) = DB::execute(connection, &sql.row_lock.create_table).await {
704            tracing::debug!(%error, "creating the lock table failed");
705        }
706        match DB::execute_bound(connection, &sql.row_lock.acquire, &[&self.table_name()]).await {
707            Err(Error::Sqlx(sqlx::Error::Database(error))) if error.is_unique_violation() => {
708                Err(Error::LockHeld {
709                    table: self.dialect.lock_table_name(),
710                })
711            }
712            result => result,
713        }
714    }

729    /// Generate migration plan using pre-fetched applied migration rows.
730    ///
731    /// This is the core plan-generation logic that performs **no database
732    /// access**. It is useful when you have already called
733    /// [`Migrator::fetch_applied_migration_from_db`] (e.g., to
734    /// display status alongside the plan) and want to avoid a redundant
735    /// database round-trip.
736    ///
737    /// If `plan` is `None`, returns all migrations in apply order. If `plan`
738    /// is `Some`, processes the plan using the supplied
739    /// `applied_migration_sql_rows`.
740    ///
741    /// # Errors
742    /// Returns an error if the migration list is empty, contains unresolved
743    /// virtual migrations, has a dependency deadlock, or if the plan
744    /// references an app/migration that does not exist.
745    pub fn generate_migration_plan(
746        &self,
747        plan: Option<&Plan>,
748        applied_migration_sql_rows: &[AppliedMigrationSqlRow],
749    ) -> MigrationVecResult<'_, DB> {
750        if self.migrations().is_empty() {
751            return Err(Error::PlanError {
752                message: "no migrations are added to the migration list".to_string(),
753            });
754        }
755        // if there is any virtual migration which is not replaced then return
756        // error since virtual migration should only be used for replacing
757        // another migration
758        if self
759            .migrations()
760            .iter()
761            .any(|migration| migration.is_virtual())
762        {
763            return Err(Error::PlanError {
764                message: "virtual migration which is not replaced is present".to_string(),
765            });
766        }
767
768        tracing::debug!("generating {:?} migration plan", plan);
769
770        let relations = Relations::new(self.migrations())?;
771        let order = relations.order()?;
772
773        // if plan is provided then modify migration list according to plan else
774        // return all migration in order of apply
775        let order = if let Some(some_plan) = plan {
776            let applied_keys: HashSet<(&str, &str)> = applied_migration_sql_rows
777                .iter()
778                .map(|row| (row.app(), row.name()))
779                .collect();
780            let applied = self
781                .migrations()
782                .iter()
783                .map(|migration| applied_keys.contains(&(migration.app(), migration.name())))
784                .collect::<Vec<_>>();
785            let last_batch_keys: HashSet<(&str, &str)> = match &some_plan.plan_type {
786                PlanType::Revert(revert_plan) if revert_plan.last_batch => {
787                    let last_batch = applied_migration_sql_rows
788                        .iter()
789                        .map(AppliedMigrationSqlRow::batch)
790                        .max();
791                    applied_migration_sql_rows
792                        .iter()
793                        .filter(|row| Some(row.batch()) == last_batch)
794                        .map(|row| (row.app(), row.name()))
795                        .collect()
796                }
797                _ => HashSet::new(),
798            };
799            relations.plan(order, &applied, some_plan, &last_batch_keys)?
800        } else {
801            order
802        };
803
804        Ok(order
805            .into_iter()
806            .map(|index| &self.migrations()[index])
807            .collect())
808    }

810    /// Ensures no pending migrations are present.
811    ///
812    /// This is useful as an application startup check to refuse serving until
813    /// all migrations are applied.
814    ///
815    /// # Errors
816    /// Returns [`Error::PendingMigrationPresent`] when at least one migration
817    /// is pending, or other errors when plan generation fails
818    pub async fn check_pending(
819        &self,
820        connection: &mut <DB as Database>::Connection,
821    ) -> Result<(), Error> {
822        let applied_rows = self.fetch_applied_migration_from_db(connection).await?;
823        let pending = self.generate_migration_plan(Some(&Plan::apply_all()), &applied_rows)?;
824        if !pending.is_empty() {
825            return Err(Error::PendingMigrationPresent);
826        }
827        Ok(())
828    }

830    /// Generate migration status using pre-fetched applied migration rows.
831    ///
832    /// This is the core status logic which performs **no database access**.
833    /// It is useful when you have already called
834    /// [`Migrator::fetch_applied_migration_from_db`] for another
835    /// purpose and want to avoid a redundant database round-trip. Returns
836    /// every migration in apply order paired with a reference to its applied
837    /// migration row when applied, or an empty list when no migrations are
838    /// added.
839    ///
840    /// # Errors
841    /// If plan generation fails
842    pub fn status<'get>(
843        &self,
844        applied_migration_sql_rows: &'get [AppliedMigrationSqlRow],
845    ) -> Result<StatusVec<'_, 'get, DB>, Error> {
846        if self.migrations().is_empty() {
847            return Ok(vec![]);
848        }
849        let applied_by_key = applied_migration_sql_rows
850            .iter()
851            .map(|row| ((row.app(), row.name()), row))
852            .collect::<HashMap<_, _>>();
853        Ok(self
854            .generate_migration_plan(None, &[])?
855            .into_iter()
856            .map(|migration| {
857                let applied = applied_by_key
858                    .get(&(migration.app(), migration.name()))
859                    .copied();
860                (migration, applied)
861            })
862            .collect())
863    }

865    /// Prune the migration table by deleting applied migration entries which
866    /// no longer have a corresponding migration in the migration list.
867    ///
868    /// This is useful for cleaning up the migration table after migrations
869    /// are removed from the codebase, since such entries would otherwise
870    /// linger in the migration table forever. Returns the number of pruned
871    /// entries.
872    ///
873    /// # Errors
874    /// If fetching or deleting applied migrations fails
875    pub async fn prune(
876        &self,
877        connection: &mut <DB as Database>::Connection,
878    ) -> Result<usize, Error> {
879        tracing::debug!("pruning unknown applied migrations");
880        let mut locked = self.lock(connection).await?;
881        // store result of pruning so that lock is unlocked before returning
882        // result
883        let result = async {
884            let connection = locked.connection();
885            let applied_migrations = self.fetch_applied_migration_from_db(connection).await?;
886            let known_migrations = self
887                .migrations()
888                .iter()
889                .map(|migration| (migration.app(), migration.name()))
890                .collect::<HashSet<_>>();
891            let mut pruned_count = 0;
892            for row in &applied_migrations {
893                if !known_migrations.contains(&(row.app(), row.name())) {
894                    let unknown_migration = (row.app().to_string(), row.name().to_string());
895                    self.delete_migration_from_db_table(connection, &unknown_migration)
896                        .await?;
897                    pruned_count += 1;
898                }
899            }
900            Ok(pruned_count)
901        }
902        .await;
903        // unlock before returning; if both prune and unlock fail, the prune
904        // error takes precedence over the unlock error.
905        let unlock_result = self.unlock(locked, result.is_ok()).await;
906        result.and_then(|count| unlock_result.map(|()| count))
907    }

909    /// Removes the migrations which started running and never finished from
910    /// the migration table, so they run again. Returns the number of removed
911    /// entries.
912    ///
913    /// Only the migration table is changed: the operations of a failed
914    /// migration ran partly, so the database has to be fixed by hand first.
915    ///
916    /// # Errors
917    /// If fetching or deleting failed migrations fails
918    pub async fn repair(
919        &self,
920        connection: &mut <DB as Database>::Connection,
921    ) -> Result<usize, Error> {
922        tracing::debug!("repairing failed migrations");
923        let mut locked = self.lock(connection).await?;
924        // store result of repairing so that lock is unlocked before returning
925        // result
926        let result = async {
927            let connection = locked.connection();
928            let applied_migrations = self.fetch_applied_migration_from_db(connection).await?;
929            let failed_count = applied_migrations
930                .iter()
931                .filter(|row| !row.success())
932                .count();
933            if failed_count > 0 {
934                DB::execute(connection, &self.dialect.sql().delete_failed_migrations).await?;
935            }
936            Ok(failed_count)
937        }
938        .await;
939        // unlock before returning; if both repair and unlock fail, the repair
940        // error takes precedence over the unlock error.
941        let unlock_result = self.unlock(locked, result.is_ok()).await;
942        result.and_then(|count| unlock_result.map(|()| count))
943    }

945    /// Records the migrations an old migrator reports as applied, skipping
946    /// ones this migrator does not know or which are already recorded.
947    /// Applied times are not preserved. Use it to move from another
948    /// migration system or to rename the migration table.
949    ///
950    /// # Errors
951    /// If sync cannot be completed
952    pub async fn sync<O>(
953        &self,
954        connection: &mut <DB as Database>::Connection,
955        old_migrator: &O,
956    ) -> Result<(), Error>
957    where
958        O: OldMigrator<DB>,
959    {
960        tracing::debug!("syncing old migrator");
961        let mut locked = self.lock(connection).await?;
962        // Use a block to ensure the lock is released before returning
963        let result = async {
964            let connection = locked.connection();
965            let old_migrator_applied_migrations =
966                old_migrator.applied_migrations(connection).await?;
967            let already_applied_migration =
968                self.fetch_applied_migration_from_db(connection).await?;
969            let known_migrations = self
970                .migrations
971                .iter()
972                .map(|migration| (migration.app(), migration.name()))
973                .collect::<HashSet<_>>();
974            let already_applied = already_applied_migration
975                .iter()
976                .map(|row| (row.app(), row.name()))
977                .collect::<HashSet<_>>();
978            // synced migrations are recorded as one batch
979            let batch = next_batch(&already_applied_migration);
980            for migration in old_migrator_applied_migrations {
981                let key = (migration.app(), migration.name());
982                if known_migrations.contains(&key) && !already_applied.contains(&key) {
983                    self.add_migration_to_db_table(
984                        connection,
985                        migration.as_ref(),
986                        std::time::Duration::ZERO,
987                        batch,
988                    )
989                    .await?;
990                }
991            }
992            Ok(())
993        }
994        .await;
995        // unlock before returning; if both sync and unlock fail, the sync
996        // error takes precedence over the unlock error.
997        let unlock_result = self.unlock(locked, result.is_ok()).await;
998        result.and(unlock_result)
999    }

1001    /// Run provided plan migrations
1002    ///
1003    /// Returns a [`RunReport`] listing every migration which was applied or
1004    /// reverted along with the time each one took. Progress is emitted as
1005    /// `tracing` events at `info` level inside a span per run and per
1006    /// migration.
1007    ///
1008    /// # Errors
1009    /// If failed to run provided plan migrations
1010    pub async fn run(
1011        &self,
1012        connection: &mut <DB as Database>::Connection,
1013        plan: &Plan,
1014    ) -> Result<RunReport, Error> {
1015        let run_span = tracing::info_span!(
1016            "sqlx_migrator_run",
1017            plan_type = ?plan.plan_type,
1018            fake = plan.fake
1019        );
1020        async {
1021            tracing::debug!("running plan {:?}", plan);
1022            let mut locked = self.lock(connection).await?;
1023            // store result of applying migration so that we can unlock lock
1024            // before returning result
1025            let result = run_plan(self, locked.connection(), plan).await;
1026            // unlock before returning; if both migration and unlock fail, the
1027            // migration error takes precedence over the unlock error.
1028            let unlock_result = self.unlock(locked, result.is_ok()).await;
1029            let report = result.and_then(|report| unlock_result.map(|()| report))?;
1030            tracing::info!(
1031                count = report.len(),
1032                elapsed_ns = report.total_elapsed().as_nanos(),
1033                "run finished"
1034            );
1035            Ok(report)
1036        }
1037        .instrument(run_span)
1038        .await
1039    }
1040}

1042/// Batch number for the migrations a run records, one more than the highest
1043/// batch in the migration table
1044fn next_batch(applied_rows: &[AppliedMigrationSqlRow]) -> i32 {
1045    applied_rows
1046        .iter()
1047        .map(AppliedMigrationSqlRow::batch)
1048        .max()
1049        .unwrap_or(0)
1050        + 1
1051}

1053/// Applies one migration and records it in the migration table. A non atomic
1054/// migration is recorded as failed before its operations run and marked
1055/// successful after the last one, so one which fails part way stays recorded
1056/// as failed instead of being run again on top of what it already changed.
1057async fn apply_migration<DB>(
1058    migrator: &Migrator<DB>,
1059    connection: &mut <DB as Database>::Connection,
1060    migration: &BoxMigration<DB>,
1061    fake: bool,
1062    batch: i32,
1063) -> Result<(), Error>
1064where
1065    DB: SupportedDatabase,
1066{
1067    let start_instant = std::time::Instant::now();
1068    let operations = migration.operations();
1069    if migration.is_atomic() {
1070        let mut transaction = connection.begin().await?;
1071        if !fake {
1072            for operation in operations {
1073                operation.up(&mut transaction).await?;
1074            }
1075        }
1076        let execution_time = if fake {
1077            std::time::Duration::ZERO
1078        } else {
1079            start_instant.elapsed()
1080        };
1081        migrator
1082            .add_migration_to_db_table(&mut transaction, migration.as_ref(), execution_time, batch)
1083            .await?;
1084        transaction.commit().await?;
1085    } else if fake {
1086        migrator
1087            .add_migration_to_db_table(
1088                connection,
1089                migration.as_ref(),
1090                std::time::Duration::ZERO,
1091                batch,
1092            )
1093            .await?;
1094    } else {
1095        migrator
1096            .start_migration_in_db_table(connection, migration.as_ref(), batch)
1097            .await?;
1098        for operation in operations {
1099            operation.up(connection).await?;
1100        }
1101        migrator
1102            .finish_migration_in_db_table(connection, migration.as_ref(), start_instant.elapsed())
1103            .await?;
1104    }
1105    Ok(())
1106}

1108/// Runs the plan while the lock is held. Split out of [`Migrator::run`] so
1109/// the lock is always released whatever the outcome.
1110async fn run_plan<DB>(
1111    migrator: &Migrator<DB>,
1112    connection: &mut <DB as Database>::Connection,
1113    plan: &Plan,
1114) -> Result<RunReport, Error>
1115where
1116    DB: SupportedDatabase,
1117{
1118    let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
1119    if let Some(failed) = applied_rows.iter().find(|row| !row.success()) {
1120        return Err(Error::FailedMigration {
1121            app: failed.app().to_string(),
1122            name: failed.name().to_string(),
1123        });
1124    }
1125    let plan_migrations = migrator.generate_migration_plan(Some(plan), &applied_rows)?;
1126    // the transaction lock holds the whole plan in one transaction
1127    if !plan.fake
1128        && migrator.dialect().lock_strategy().in_transaction()
1129        && plan_migrations
1130            .iter()
1131            .any(|migration| !migration.is_atomic())
1132    {
1133        return Err(Error::PlanError {
1134            message: "the transaction lock runs the plan in one transaction so every migration \
1135                      must be atomic"
1136                .to_string(),
1137        });
1138    }
1139    // migrations applied by this run share the next batch number
1140    let batch = next_batch(&applied_rows);
1141    let mut report = RunReport {
1142        apply: matches!(plan.plan_type, PlanType::Apply),
1143        fake: plan.fake,
1144        records: Vec::with_capacity(plan_migrations.len()),
1145    };
1146    if plan_migrations.is_empty() {
1147        tracing::info!("no migration to run");
1148    }
1149    for migration in plan_migrations {
1150        let migration_span = tracing::info_span!(
1151            "sqlx_migrator_migration",
1152            app = migration.app(),
1153            name = migration.name()
1154        );
1155        let elapsed = async {
1156            let start_instant = std::time::Instant::now();
1157            match plan.plan_type {
1158                PlanType::Apply => {
1159                    tracing::info!("applying migration");
1160                    apply_migration(migrator, connection, migration, plan.fake, batch).await?;
1161                }
1162                PlanType::Revert(_) => {
1163                    tracing::info!("reverting migration");
1164
1165                    // Reverse operation since last applied operation need
1166                    // to be reverted first
1167                    let mut operations = migration.operations();
1168                    operations.reverse();
1169
1170                    if migration.is_atomic() {
1171                        let mut transaction = connection.begin().await?;
1172                        if !plan.fake {
1173                            for operation in operations {
1174                                operation.down(&mut transaction).await?;
1175                            }
1176                        }
1177                        migrator
1178                            .delete_migration_from_db_table(&mut transaction, migration.as_ref())
1179                            .await?;
1180                        transaction.commit().await?;
1181                    } else {
1182                        if !plan.fake {
1183                            for operation in operations {
1184                                operation.down(connection).await?;
1185                            }
1186                        }
1187                        migrator
1188                            .delete_migration_from_db_table(connection, migration.as_ref())
1189                            .await?;
1190                    }
1191                }
1192            }
1193            let elapsed = start_instant.elapsed();
1194            tracing::info!(elapsed_ns = elapsed.as_nanos(), "migration finished");
1195            Ok::<_, Error>(elapsed)
1196        }
1197        .instrument(migration_span)
1198        .await?;
1199        report.records.push(RunRecord {
1200            app: migration.app().to_string(),
1201            name: migration.name().to_string(),
1202            elapsed,
1203        });
1204    }
1205    Ok(report)
