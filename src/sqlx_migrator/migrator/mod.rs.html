190/// Struct that determines the type of migration plan to execute.
191///
192/// A [`Plan`] can specify whether to apply or revert migrations, and may target
193/// all migrations, specific migrations, or a limited number of migrations.
194#[derive(Debug)]
195pub struct Plan {
196    #[expect(
197        clippy::struct_field_names,
198        reason = "type is a keyword so it cannot be used"
199    )]
200    plan_type: PlanType,
201    app_migration: Option<(String, Option<String>)>,
202    count: Option<usize>,
203    fake: bool,
204}
205
206impl Plan {
207    fn new(
208        plan_type: PlanType,
209        app_migration: Option<(String, Option<String>)>,
210        count: Option<usize>,
211    ) -> Self {
212        Self {
213            plan_type,
214            app_migration,
215            count,
216            fake: false,
217        }
218    }
219
220    /// Sets the plan as a "fake" plan.
221    ///
222    /// When the plan is marked as fake, the migration status is updated to
223    /// either "applied" or "reverted" without actually performing any
224    /// migration operations. This is useful for scenarios where you want to
225    /// simulate the effect of applying or reverting a migration, but
226    /// without making changes to the database.
227    ///
228    /// By default, the `fake` flag is set to `false`, and the migration
229    /// operations are executed as expected.
230    #[must_use]
231    pub fn fake(mut self, fake: bool) -> Self {
232        self.fake = fake;
233        self
234    }
235
236    /// Creates a new plan to apply all migrations.
237    #[must_use]
238    pub fn apply_all() -> Self {
239        Self::new(PlanType::Apply, None, None)
240    }
241
242    /// Creates a new plan to apply a specific migration by name. If migration
243    /// name is not provided it will apply all migrations of the app
244    #[must_use]
245    pub fn apply_name(app: &str, name: Option<&str>) -> Self {
246        Self::new(
247            PlanType::Apply,
248            Some((app.to_string(), name.map(ToString::to_string))),
249            None,
250        )
251    }
252
253    /// Creates a new plan to apply a limited number of migrations.
254    #[must_use]
255    pub fn apply_count(count: usize) -> Self {
256        Self::new(PlanType::Apply, None, Some(count))
257    }
258
259    /// Creates a new plan to revert all migrations.
260    #[must_use]
261    pub fn revert_all() -> Self {
262        Self::new(PlanType::Revert, None, None)
263    }
264
265    /// Creates a new plan to revert a specific migration by name. If migration
266    /// name is not provided it will revert all migrations of the app
267    #[must_use]
268    pub fn revert_name(app: &str, name: Option<&str>) -> Self {
269        Self::new(
270            PlanType::Revert,
271            Some((app.to_string(), name.map(ToString::to_string))),
272            None,
273        )
274    }
275
276    /// Creates a new plan to revert a limited number of migrations.
277    #[must_use]
278    pub fn revert_count(count: usize) -> Self {
279        Self::new(PlanType::Revert, None, Some(count))
280    }

380/// The [`Info`] trait provides database-agnostic methods for managing
381/// migrations and interacting with migration states.
382pub trait Info<DB> {
383    /// Returns a reference to the list of migrations.
384    fn migrations(&self) -> &[BoxMigration<DB>];
385
386    /// Returns a mutable reference to the list of migrations.
387    fn migrations_mut(&mut self) -> &mut Vec<BoxMigration<DB>>;
388
389    /// Adds a list of migrations to the migrator.
390    ///
391    /// This method accepts a vector of migrations and adds each one
392    /// individually to ensure proper handling of migration relationships
393    /// and duplicates.
394    ///
395    /// # Errors
396    /// If migration is added with same app and name but inconsistent value i.e
397    /// its parents, run before, replaces and is atomic differ and do not have
398    /// same number of operation
399    fn add_migrations(&mut self, migrations: Vec<BoxMigration<DB>>) -> Result<(), Error> {
400        for migration in migrations {
401            self.add_migration(migration)?;
402        }
403        Ok(())
404    }
405
406    /// Adds a single migration to the migrator.
407    ///
408    /// # Errors
409    /// If migration is added with same app and name but inconsistent value i.e
410    /// its parents, run before, replaces and is atomic differ and do not have
411    /// same number of operation
412    fn add_migration(&mut self, migration: BoxMigration<DB>) -> Result<(), Error> {
413        if migration.is_virtual() {
414            // Virtual migrations must not have any fields other than app/name
415            if !migration.parents().is_empty()
416                || !migration.operations().is_empty()
417                || !migration.replaces().is_empty()
418                || !migration.run_before().is_empty()
419            {
420                return Err(Error::InvalidVirtualMigration);
421            }
422            // Only add virtual migration if not already present
423            if self.migrations().contains(&migration) {
424                return Ok(());
425            }
426            self.migrations_mut().push(migration);
427        } else {
428            let mut skip_add = false;
429            if let Some((migration_index, found_migration)) = self
430                .migrations()
431                .iter()
432                .enumerate()
433                .find(|(_, elem)| elem == &&migration)
434            {
435                if found_migration.is_virtual() {
436                    // Replace the placeholder virtual migration with the
437                    // concrete one
438                    self.migrations_mut().remove(migration_index);
439                } else if found_migration.parents() != migration.parents()
440                    || found_migration.operations().len() != migration.operations().len()
441                    || found_migration.replaces() != migration.replaces()
442                    || found_migration.run_before() != migration.run_before()
443                    || found_migration.is_atomic() != migration.is_atomic()
444                {
445                    // Non-virtual duplicate with different definition
446                    return Err(Error::InconsistentMigration {
447                        app: migration.app().to_string(),
448                        name: migration.name().to_string(),
449                    });
450                } else {
451                    // Non-virtual consistent duplicate — already registered
452                    skip_add = true;
453                }
454            }
455
456            if !skip_add {
457                // Add this migration and recursively register its parents,
458                // replaces, and run_before migrations
459                let migration_parents = migration.parents();
460                let migration_replaces = migration.replaces();
461                let migration_run_before = migration.run_before();
462
463                self.migrations_mut().push(migration);
464
465                for parent in migration_parents {
466                    self.add_migration(parent)?;
467                }
468                for replace in migration_replaces {
469                    self.add_migration(replace)?;
470                }
471                for run_before in migration_run_before {
472                    self.add_migration(run_before)?;
473                }
474            }
475        }
476        Ok(())
477    }
478}

480/// The [`DatabaseOperation`] trait defines a set of methods for performing
481/// operations related to migration management on the database.
482///
483/// This trait is typically implemented for a database to support migration
484/// operations, such as ensuring the migration table exists, adding or
485/// removing migrations from the table, and locking the database during
486/// migration processes.
487///
488/// # Security
489/// When implementing this trait, it is crucial to ensure that all database
490/// interactions are performed securely to prevent SQL injection and other
491/// vulnerabilities. Always use parameterized queries and avoid directly
492/// interpolating user input into SQL statements. If your implementation
493/// involves dynamic SQL generation, make sure to properly sanitize and validate
494/// all inputs.
495#[async_trait::async_trait]
496pub trait DatabaseOperation<DB>
497where
498    DB: Database,
499{
500    /// Ensure migration table is created before running migrations. If not
501    /// create one
502    async fn ensure_migration_table_exists(
503        &self,
504        connection: &mut <DB as Database>::Connection,
505    ) -> Result<(), Error>;
506
507    /// Drop migration table if migration table exists
508    async fn drop_migration_table_if_exists(
509        &self,
510        connection: &mut <DB as Database>::Connection,
511    ) -> Result<(), Error>;
512
513    /// Adds a migration record to the migration table in the database.
514    async fn add_migration_to_db_table(
515        &self,
516        connection: &mut <DB as Database>::Connection,
517        migration: &dyn Migration<DB>,
518    ) -> Result<(), Error>;
519
520    /// Removes a migration record from the migration table in the database.
521    async fn delete_migration_from_db_table(
522        &self,
523        connection: &mut <DB as Database>::Connection,
524        migration: &dyn Migration<DB>,
525    ) -> Result<(), Error>;
526
527    /// Fetches the list of applied migrations from the migration table in the
528    /// database.
529    async fn fetch_applied_migration_from_db(
530        &self,
531        connection: &mut <DB as Database>::Connection,
532    ) -> Result<Vec<AppliedMigrationSqlRow>, Error>;
533
534    /// Lock database while doing migrations so no two migrations run together
535    async fn lock(&self, connection: &mut <DB as Database>::Connection) -> Result<(), Error>;
536
537    /// Unlock locked database
538    async fn unlock(&self, connection: &mut <DB as Database>::Connection) -> Result<(), Error>;
539
540    /// Whether the database can roll back DDL statements inside a
541    /// transaction.
542    ///
543    /// MySQL and MariaDB commit implicitly on most DDL statements, so an
544    /// atomic migration on those backends cannot be rolled back as a unit if
545    /// one of its operations fails. When this returns `false` the migrator
546    /// still wraps atomic migrations in a transaction, but logs a warning so
547    /// the limitation is visible.
548    ///
549    /// The default implementation returns `true`.
550    fn supports_transactional_ddl(&self, _connection: &<DB as Database>::Connection) -> bool {
551        true
552    }
553}

555/// populate replace hash map recursively
556fn populate_replace_recursive<'populate, DB>(
557    replace_hash_map: &mut HashMap<&'populate BoxMigration<DB>, Vec<&'populate BoxMigration<DB>>>,
558    key: &'populate BoxMigration<DB>,
559    value: &'populate BoxMigration<DB>,
560) -> Result<(), Error> {
561    // protect against a case where two migration replaces each other
562    if key == value {
563        return Err(Error::PlanError {
564            message: "two migrations replace each other".to_string(),
565        });
566    }
567    let replace_hash_map_vec = replace_hash_map.entry(key).or_default();
568    if !replace_hash_map_vec.contains(&value) {
569        replace_hash_map_vec.push(value);
570    }
571    // Clone only the inner Vec (cheap: Vec of references) to release the
572    // mutable borrow on replace_hash_map before the recursive call.
573    let grand_values = replace_hash_map.get(value).cloned();
574    if let Some(grand_values) = grand_values {
575        for grand_value in grand_values {
576            populate_replace_recursive(replace_hash_map, key, grand_value)?;
577        }
578    }
579    Ok(())
580}

582/// Narrows `migration_list` down to only the migrations that are required to
583/// apply or revert the specific set of *target* migrations supplied in
584/// `with_list`.
585///
586/// When a user targets a particular app/migration (e.g. `--app foo
587/// --migration bar`), the full ordered list already contains every migration in
588/// dependency order. This function keeps only the subset that is actually
589/// needed to safely apply or revert the requested targets — discarding
590/// unrelated migrations from other apps or branches.
591fn only_related_migration<DB>(
592    migration_list: &mut MigrationVec<'_, DB>,
593    with_list: Vec<&BoxMigration<DB>>,
594    plan_type: &PlanType,
595    replace_children: &HashMap<&BoxMigration<DB>, Vec<&BoxMigration<DB>>>,
596) {
597    use std::collections::VecDeque;
598
599    // `related` is a membership set only; BFS visitation order is kept by
600    // `queue`, so a HashSet keeps the inner-loop `contains` checks O(1).
601    let mut related: HashSet<&BoxMigration<DB>> = HashSet::new();
602    let mut queue: VecDeque<&BoxMigration<DB>> = VecDeque::new();
603
604    // Seed the BFS with the explicitly requested targets
605    for with in with_list {
606        if related.insert(with) {
607            queue.push_back(with);
608        }
609    }
610
611    // Expand one hop at a time until no new migrations can be reached.
612    while let Some(current) = queue.pop_front() {
613        // These depend only on `current`, so compute them once per BFS node
614        // instead of once per (current, m) pair. `parents()`/`run_before()`
615        // each allocate a fresh Vec, so hoisting avoids repeated allocations.
616        let current_parents = current.parents();
617        let current_run_before = current.run_before();
618        let current_replaced = replace_children.get(current);
619        for &m in migration_list.iter() {
620            if related.contains(&m) {
621                continue;
622            }
623            let should_include = match plan_type {
624                PlanType::Apply => {
625                    // m is a direct parent of current, or a parent of any
626                    // migration current transitively replaces (inherited edge).
627                    let is_parent = current_parents.iter().any(|p| p.as_ref() == m.as_ref())
628                        || current_replaced.is_some_and(|children| {
629                            children.iter().any(|child| {
630                                child.parents().iter().any(|p| p.as_ref() == m.as_ref())
631                            })
632                        });
633
634                    // current (or any migration it replaces) is in m's
635                    // run_before, meaning m must precede
636                    // current.
637                    let is_run_before = m
638                        .run_before()
639                        .iter()
640                        .any(|rb| rb.as_ref() == current.as_ref())
641                        || current_replaced.is_some_and(|children| {
642                            children.iter().any(|child| {
643                                m.run_before()
644                                    .iter()
645                                    .any(|rb| rb.as_ref() == child.as_ref())
646                            })
647                        });
648
649                    is_parent || is_run_before
650                }
651                PlanType::Revert => {
652                    // Collect what m transitively replaces.
653                    let m_replaced = replace_children.get(m);
654
655                    // current is a direct parent of m, or a parent of any
656                    // migration m transitively replaces (inherited edge).
657                    let is_parent = m.parents().iter().any(|p| p.as_ref() == current.as_ref())
658                        || m_replaced.is_some_and(|children| {
659                            children.iter().any(|child| {
660                                child
661                                    .parents()
662                                    .iter()
663                                    .any(|p| p.as_ref() == current.as_ref())
664                            })
665                        });
666
667                    // m (or any migration m replaces) is in current's
668                    // run_before, meaning current runs
669                    // before m, so m must be reverted before current.
670                    let is_run_before = current_run_before
671                        .iter()
672                        .any(|rb| rb.as_ref() == m.as_ref())
673                        || m_replaced.is_some_and(|children| {
674                            children.iter().any(|child| {
675                                current_run_before
676                                    .iter()
677                                    .any(|rb| rb.as_ref() == child.as_ref())
678                            })
679                        });
680
681                    is_parent || is_run_before
682                }
683            };
684            if should_include {
685                related.insert(m);
686                queue.push_back(m);
687            }
688        }
689    }
690
691    migration_list.retain(|m| related.contains(m));
692}

694/// Process plan to provided migrations list
695fn process_plan<'process, DB>(
696    migration_list: &mut MigrationVec<'process, DB>,
697    applied_migrations: &HashSet<&'process BoxMigration<DB>>,
698    plan: &Plan,
699    replace_children: &HashMap<&BoxMigration<DB>, Vec<&BoxMigration<DB>>>,
700    all_migrations: &[BoxMigration<DB>],
701) -> Result<(), Error>
702where
703    DB: Database,
704{
705    // Modify migration list according to plan type
706    match plan.plan_type {
707        PlanType::Apply => {
708            migration_list.retain(|migration| !applied_migrations.contains(migration));
709        }
710        PlanType::Revert => {
711            migration_list.retain(|migration| applied_migrations.contains(migration));
712            migration_list.reverse();
713        }
714    }
715
716    if let Some((app, migration_name)) = &plan.app_migration {
717        // Find position of last migration which matches condition of provided
718        // app and migration name. A target that exists in the full migration
719        // list but not in the processed list has nothing to do (it is already
720        // applied/reverted or replaced), so the plan is empty rather than an
721        // error — this keeps targeted apply/revert idempotent. Only a target
722        // that doesn't exist at all is an error.
723        let position = if let Some(name) = migration_name {
724            let pos = migration_list
725                .iter()
726                .rposition(|migration| migration.app() == app && migration.name() == name);
727            if pos.is_none() {
728                if !all_migrations
729                    .iter()
730                    .any(|migration| migration.app() == app)
731                {
732                    return Err(Error::PlanError {
733                        message: format!("app {app} doesn't exist"),
734                    });
735                }
736                if !all_migrations
737                    .iter()
738                    .any(|migration| migration.app() == app && migration.name() == name)
739                {
740                    return Err(Error::PlanError {
741                        message: format!("migration {name} doesn't exist for app {app}"),
742                    });
743                }
744            }
745            pos
746        } else {
747            let pos = migration_list
748                .iter()
749                .rposition(|migration| migration.app() == app);
750            if pos.is_none()
751                && !all_migrations
752                    .iter()
753                    .any(|migration| migration.app() == app)
754            {
755                return Err(Error::PlanError {
756                    message: format!("app {app} doesn't exist"),
757                });
758            }
759            pos
760        };
761        let Some(position) = position else {
762            migration_list.clear();
763            return Ok(());
764        };
765        migration_list.truncate(position + 1);
766        let with_list = if migration_name.is_some() {
767            vec![migration_list[position]]
768        } else {
769            migration_list
770                .iter()
771                .filter(|pos_migration| pos_migration.app() == app)
772                .copied()
773                .collect::<Vec<_>>()
774        };
775        only_related_migration(migration_list, with_list, &plan.plan_type, replace_children);
776    } else if let Some(count) = plan.count {
777        if count == 0 {
778            return Err(Error::PlanError {
779                message: "count must be greater than 0".to_string(),
780            });
781        }
782        let actual_len = migration_list.len();
783        if count > actual_len {
784            return Err(Error::PlanError {
785                message: format!(
786                    "passed count value is larger than migration length: {actual_len}"
787                ),
788            });
789        }
790        migration_list.truncate(count);
791    }
792    Ok(())
793}

795/// Resolve a replaced migration to the concrete migration registered in the
796/// list.
797///
798/// A virtual migration is only a placeholder for a real migration sharing the
799/// same app/name, so when `child` is virtual the matching non-virtual migration
800/// must be located; otherwise `child` itself is returned.
801fn resolve_replaced_migration<'resolve, DB>(
802    migrations: &'resolve [BoxMigration<DB>],
803    child: &'resolve BoxMigration<DB>,
804) -> Result<&'resolve BoxMigration<DB>, Error> {
805    if child.is_virtual() {
806        migrations
807            .iter()
808            .find(|&search_migration| search_migration == child)
809            .ok_or(Error::PlanError {
810                message: "Failed finding non virtual migration for virtual migration".to_string(),
811            })
812    } else {
813        Ok(child)
814    }
815}

817// get all replaces migration recursively for a migration
818fn get_recursive_replaces<'get, DB>(
819    hash_map: &'get HashMap<BoxMigration<DB>, &'get BoxMigration<DB>>,
820    val: &'get BoxMigration<DB>,
821) -> Vec<&'get BoxMigration<DB>> {
822    let mut recursive_vec = vec![val];
823    if let Some(&parent) = hash_map.get(val) {
824        recursive_vec.extend(get_recursive_replaces(hash_map, parent));
825    }
826    recursive_vec
827}

829/// The [`Migrate`] trait defines methods to manage and apply database
830/// migrations according to a given plan.
831///
832/// This trait combines the functionalities of the [`Info`] and
833/// [`DatabaseOperation`] traits, providing a full set of migration
834/// capabilities. All methods have default implementations, meaning no explicit
835/// implementation is required. Additionally, all methods are database-agnostic.
836#[async_trait::async_trait]
837pub trait Migrate<DB>: Info<DB> + DatabaseOperation<DB> + Send + Sync
838where
839    DB: Database,
840{
841    /// Generate migration plan using pre-fetched applied migration rows.
842    ///
843    /// This is the core plan-generation logic that performs **no database
844    /// access**. It is useful when you have already called
845    /// [`DatabaseOperation::fetch_applied_migration_from_db`] (e.g., to
846    /// display status alongside the plan) and want to avoid a redundant
847    /// database round-trip.
848    ///
849    /// If `plan` is `None`, returns all migrations in apply order. If `plan`
850    /// is `Some`, processes the plan using the supplied
851    /// `applied_migration_sql_rows`.
852    ///
853    /// # Errors
854    /// Returns an error if the migration list is empty, contains unresolved
855    /// virtual migrations, has a dependency deadlock, or if the plan
856    /// references an app/migration that does not exist.
857    #[expect(clippy::too_many_lines)]
858    fn generate_migration_plan(
859        &self,
860        plan: Option<&Plan>,
861        applied_migration_sql_rows: &[AppliedMigrationSqlRow],
862    ) -> MigrationVecResult<'_, DB> {
863        if self.migrations().is_empty() {
864            return Err(Error::PlanError {
865                message: "no migrations are added to the migration list".to_string(),
866            });
867        }
868        // if there is any virtual migration which is not replaced then return
869        // error since virtual migration should only be used for replacing
870        // another migration
871        if self
872            .migrations()
873            .iter()
874            .any(|migration| migration.is_virtual())
875        {
876            return Err(Error::PlanError {
877                message: "virtual migration which is not replaced is present".to_string(),
878            });
879        }
880
881        tracing::debug!("generating {:?} migration plan", plan);
882
883        // Hashmap which contains key as child migration and value as parent
884        // migration Child migration is migration which is replaced by
885        // parent migration. One migration can only be replaced by one
886        // migration but one migration can replace multiple migration so
887        // we can have multiple child for one parent but one child
888        // can only have one parent
889        let mut replaces_child_parent_hash_map = HashMap::new();
890
891        for parent_migration in self.migrations() {
892            for child_migration in parent_migration.replaces() {
893                let child_name = format!("{}:{}", child_migration.app(), child_migration.name());
894                if replaces_child_parent_hash_map
895                    .insert(child_migration, parent_migration)
896                    .is_some()
897                {
898                    return Err(Error::PlanError {
899                        message: format!("migration {child_name} replaced multiple times"),
900                    });
901                }
902            }
903        }
904
905        // Hashmap which contains key as migration and value is vector of
906        // migration which are children of this migration due to
907        // replace. One migration can have multiple children
908        let mut replace_children = HashMap::<_, Vec<_>>::new();
909        // in first loop add direct children of parent due to replace
910        for (child, &parent) in &replaces_child_parent_hash_map {
911            let children_migration = resolve_replaced_migration(self.migrations(), child)?;
912            replace_children
913                .entry(parent)
914                .or_default()
915                .push(children_migration);
916        }
917        // in second loop add recursive children of parent due to replace
918        for (child, &parent) in &replaces_child_parent_hash_map {
919            let children_migration = resolve_replaced_migration(self.migrations(), child)?;
920            populate_replace_recursive(&mut replace_children, parent, children_migration)?;
921        }
922
923        // Hashmap which contains key as migration and value is vector of
924        // migration which should run before this migration. One
925        // migration can have multiple run before migration
926        let mut run_before_child_parent_hash_map = HashMap::<_, Vec<_>>::new();
927
928        for parent_migration in self.migrations() {
929            for run_before_migration in parent_migration.run_before() {
930                run_before_child_parent_hash_map
931                    .entry(run_before_migration)
932                    .or_default()
933                    .push(parent_migration);
934            }
935        }
936
937        let mut migration_list = Vec::new();
938        // HashSet to keep track of migration which are already added to
939        // migration list to avoid adding same migration multiple times
940        // to migration list since one migration can be parent, run
941        // before or replace multiple migration
942        let mut migration_set = HashSet::new();
943
944        // keep looping until all migration are added to migration list. In each
945        // loop check if any migration can be added to migration list or
946        // not. A migration can be added if all its parents are already
947        // added to migration list
948        let original_migration_length = self.migrations().len();
949        while migration_list.len() != original_migration_length {
950            let loop_initial_migration_list_length = migration_list.len();
951            for migration in self.migrations() {
952                // check if all parents and run before migration are already
953                // added to migration list and if it replaces
954                // any migration then that migration
955                // should be added to migration list as well before adding this
956                // migration to migration list. Also if this
957                // migration have children due to replace
958                // then their parents and run before should be added to
959                // migration list before adding this migration
960                // to migration list
961                let all_required_added = !migration_set.contains(migration)
962                    && migration
963                        .parents()
964                        .iter()
965                        .all(|p| migration_set.contains(p))
966                    && run_before_child_parent_hash_map
967                        .get(migration)
968                        .unwrap_or(&vec![])
969                        .iter()
970                        .all(|rb| migration_set.contains(rb))
971                    && replaces_child_parent_hash_map
972                        .get(migration)
973                        .is_none_or(|r| migration_set.contains(r))
974                    && replace_children.get(migration).is_none_or(|children| {
975                        // if children are present then their parents and run
976                        // before should be
977                        // added to migration list before adding replace
978                        // migration
979                        children.iter().all(|&child| {
980                            child.parents().iter().all(|p| migration_set.contains(p))
981                                && run_before_child_parent_hash_map
982                                    .get(child)
983                                    .unwrap_or(&vec![])
984                                    .iter()
985                                    .all(|rb| migration_set.contains(rb) || children.contains(rb))
986                        })
987                    });
988                if all_required_added {
989                    migration_list.push(migration);
990                    migration_set.insert(migration);
991                }
992            }
993
994            // if no migration is added in this loop then it means there is a
995            // deadlock and we cannot proceed further
996            if loop_initial_migration_list_length == migration_list.len() {
997                return Err(Error::PlanError {
998                    message: "reached deadlock stage during plan generation".to_string(),
999                });
1000            }
1001        }
1002
1003        // if plan is provided then modify migration list according to plan else
1004        // return all migration in order of apply
1005        if let Some(some_plan) = plan {
1006            // Index the applied rows by (app, name) once so matching each
1007            // migration is an O(1) lookup rather than an O(rows) rescan, and
1008            // keep the result in a set for O(1) membership tests below.
1009            let applied_row_keys: HashSet<(&str, &str)> = applied_migration_sql_rows
1010                .iter()
1011                .map(|row| (row.app(), row.name()))
1012                .collect();
1013            let mut applied_migrations = HashSet::new();
1014            for migration in self.migrations() {
1015                if applied_row_keys.contains(&(migration.app(), migration.name())) {
1016                    applied_migrations.insert(migration);
1017                }
1018            }
1019
1020            // Check if any child migration is applied before its parent
1021            // migration according to parents and run before field.
1022            // If yes then return error. Iterate `self.migrations()`
1023            // (not the set) so the reported error is deterministic
1024            // when multiple migrations violate the constraint.
1025            for migration in self.migrations() {
1026                if !applied_migrations.contains(&migration) {
1027                    continue;
1028                }
1029                let mut parents = vec![];
1030                if let Some(run_before_list) = run_before_child_parent_hash_map.get(migration) {
1031                    for &run_before in run_before_list {
1032                        parents.push(run_before);
1033                    }
1034                }
1035                let main_parents = migration.parents();
1036                for parent in &main_parents {
1037                    parents.push(parent);
1038                }
1039                for parent in parents {
1040                    let recursive_vec =
1041                        get_recursive_replaces(&replaces_child_parent_hash_map, parent);
1042                    if !applied_migrations
1043                        .iter()
1044                        .any(|applied| recursive_vec.contains(applied))
1045                    {
1046                        return Err(Error::PlanError {
1047                            message: format!(
1048                                "child migration {}:{} applied before its parent migration {}:{}",
1049                                migration.app(),
1050                                migration.name(),
1051                                parent.app(),
1052                                parent.name()
1053                            ),
1054                        });
1055                    }
1056                }
1057            }
1058
1059            // Check if any migration and its replaces are applied together or
1060            // not. Collect all removals first to avoid cloning
1061            // migration_list.
1062            let mut to_remove_replacers: Vec<&BoxMigration<DB>> = vec![];
1063            let mut to_remove_replaced: Vec<&BoxMigration<DB>> = vec![];
1064
1065            for &migration in &migration_list {
1066                // Check if this migration have any children due to replace
1067                if let Some(children) = replace_children.get(&migration) {
1068                    // Check if any one of replaced migration is applied or not
1069                    let replaces_applied = children
1070                        .iter()
1071                        .any(|&replace_migration| applied_migrations.contains(&replace_migration));
1072
1073                    // If replaces migration is applied then we cannot apply
1074                    // this migration If replaces migration
1075                    // is not applied then we can remove all replaced
1076                    // migration from migration list since this migration will
1077                    // apply in place of them
1078                    if replaces_applied {
1079                        // Errors out if this migration is also applied since
1080                        // both migration and its
1081                        // replaces cannot be applied together
1082                        if applied_migrations.contains(&migration) {
1083                            return Err(Error::PlanError {
1084                                message: format!(
1085                                    "migration {}:{} and its replaces are applied together",
1086                                    migration.app(),
1087                                    migration.name(),
1088                                ),
1089                            });
1090                        }
1091                        to_remove_replacers.push(migration);
1092                    } else {
1093                        // remove all replaced migration from migration list
1094                        // since this migration will
1095                        // apply in place of them
1096                        to_remove_replaced.extend(children);
1097                    }
1098                }
1099            }
1100
1101            migration_list.retain(|&m| {
1102                let list_contains_migration =
1103                    to_remove_replacers.contains(&m) || to_remove_replaced.contains(&m);
1104                !list_contains_migration
1105            });
1106
1107            process_plan(
1108                &mut migration_list,
1109                &applied_migrations,
1110                some_plan,
1111                &replace_children,
1112                self.migrations(),
1113            )?;
1114        }
1115
1116        Ok(migration_list)
1117    }
1118
1119    /// Ensures no pending migrations are present.
1120    ///
1121    /// This is useful as an application startup check to refuse serving until
1122    /// all migrations are applied.
1123    ///
1124    /// # Errors
1125    /// Returns [`Error::PendingMigrationPresent`] when at least one migration
1126    /// is pending, or other errors when plan generation fails
1127    async fn check_pending(
1128        &self,
1129        connection: &mut <DB as Database>::Connection,
1130    ) -> Result<(), Error> {
1131        self.ensure_migration_table_exists(connection).await?;
1132        let applied_rows = self.fetch_applied_migration_from_db(connection).await?;
1133        let pending = self.generate_migration_plan(Some(&Plan::apply_all()), &applied_rows)?;
1134        if !pending.is_empty() {
1135            return Err(Error::PendingMigrationPresent);
1136        }
1137        Ok(())
1138    }

1140    /// Generate migration status using pre-fetched applied migration rows.
1141    ///
1142    /// This is the core status logic which performs **no database access**.
1143    /// It is useful when you have already called
1144    /// [`DatabaseOperation::fetch_applied_migration_from_db`] for another
1145    /// purpose and want to avoid a redundant database round-trip. Returns
1146    /// every migration in apply order paired with a reference to its applied
1147    /// migration row when applied, or an empty list when no migrations are
1148    /// added.
1149    ///
1150    /// # Errors
1151    /// If plan generation fails
1152    fn status<'get>(
1153        &self,
1154        applied_migration_sql_rows: &'get [AppliedMigrationSqlRow],
1155    ) -> Result<StatusVec<'_, 'get, DB>, Error> {
1156        if self.migrations().is_empty() {
1157            return Ok(vec![]);
1158        }
1159        let applied_by_key = applied_migration_sql_rows
1160            .iter()
1161            .map(|row| ((row.app(), row.name()), row))
1162            .collect::<HashMap<_, _>>();
1163        Ok(self
1164            .generate_migration_plan(None, &[])?
1165            .into_iter()
1166            .map(|migration| {
1167                let applied = applied_by_key
1168                    .get(&(migration.app(), migration.name()))
1169                    .copied();
1170                (migration, applied)
1171            })
1172            .collect())
1173    }

1175    /// Prune the migration table by deleting applied migration entries which
1176    /// no longer have a corresponding migration in the migration list.
1177    ///
1178    /// This is useful for cleaning up the migration table after migrations
1179    /// are removed from the codebase, since such entries would otherwise
1180    /// linger in the migration table forever. Returns the number of pruned
1181    /// entries.
1182    ///
1183    /// # Errors
1184    /// If fetching or deleting applied migrations fails
1185    async fn prune(&self, connection: &mut <DB as Database>::Connection) -> Result<usize, Error> {
1186        tracing::debug!("pruning unknown applied migrations");
1187        self.lock(connection).await?;
1188        // store result of pruning so that lock is unlocked before returning
1189        // result
1190        let result = async {
1191            self.ensure_migration_table_exists(connection).await?;
1192            let applied_migrations = self.fetch_applied_migration_from_db(connection).await?;
1193            let known_migrations = self
1194                .migrations()
1195                .iter()
1196                .map(|migration| (migration.app(), migration.name()))
1197                .collect::<HashSet<_>>();
1198            let mut pruned_count = 0;
1199            for row in &applied_migrations {
1200                if !known_migrations.contains(&(row.app(), row.name())) {
1201                    let unknown_migration = (row.app().to_string(), row.name().to_string());
1202                    self.delete_migration_from_db_table(connection, &unknown_migration)
1203                        .await?;
1204                    pruned_count += 1;
1205                }
1206            }
1207            Ok(pruned_count)
1208        }
1209        .await;
1210        // unlock before returning; if both prune and unlock fail, the prune
1211        // error takes precedence over the unlock error.
1212        let unlock_result = self.unlock(connection).await;
1213        result.and_then(|count| unlock_result.map(|()| count))
1214    }
1215
1216    /// Run provided plan migrations
1217    ///
1218    /// Returns a [`RunReport`] listing every migration which was applied or
1219    /// reverted along with the time each one took. Progress is emitted as
1220    /// `tracing` events at `info` level inside a span per run and per
1221    /// migration.
1222    ///
1223    /// # Errors
1224    /// If failed to run provided plan migrations
1225    async fn run(
1226        &self,
1227        connection: &mut <DB as Database>::Connection,
1228        plan: &Plan,
1229    ) -> Result<RunReport, Error> {
1230        let run_span = tracing::info_span!(
1231            "sqlx_migrator_run",
1232            plan_type = ?plan.plan_type,
1233            fake = plan.fake
1234        );
1235        async {
1236            tracing::debug!("running plan {:?}", plan);
1237            self.lock(connection).await?;
1238            // store result of applying migration so that we can unlock lock
1239            // before returning result
1240            let result = run_plan(self, connection, plan).await;
1241            // unlock before returning; if both migration and unlock fail, the
1242            // migration error takes precedence over the unlock error.
1243            let unlock_result = self.unlock(connection).await;
1244            let report = result.and_then(|report| unlock_result.map(|()| report))?;
1245            tracing::info!(
1246                count = report.len(),
1247                elapsed_ms = report.total_elapsed().as_millis(),
1248                "run finished"
1249            );
1250            Ok(report)
1251        }
1252        .instrument(run_span)
1253        .await
1254    }
1255}

1257/// Runs the plan while the lock is held. Split out of [`Migrate::run`] so
1258/// the lock is always released whatever the outcome.
1259async fn run_plan<DB, M>(
1260    migrator: &M,
1261    connection: &mut <DB as Database>::Connection,
1262    plan: &Plan,
1263) -> Result<RunReport, Error>
1264where
1265    DB: Database,
1266    M: Migrate<DB> + ?Sized,
1267{
1268    migrator.ensure_migration_table_exists(connection).await?;
1269    let applied_rows = migrator.fetch_applied_migration_from_db(connection).await?;
1270    let plan_migrations = migrator.generate_migration_plan(Some(plan), &applied_rows)?;
1271    let supports_transactional_ddl = migrator.supports_transactional_ddl(connection);
1272    let mut report = RunReport {
1273        apply: matches!(plan.plan_type, PlanType::Apply),
1274        fake: plan.fake,
1275        records: Vec::with_capacity(plan_migrations.len()),
1276    };
1277    if plan_migrations.is_empty() {
1278        tracing::info!("no migration to run");
1279    }
1280    for migration in plan_migrations {
1281        let migration_span = tracing::info_span!(
1282            "sqlx_migrator_migration",
1283            app = migration.app(),
1284            name = migration.name()
1285        );
1286        let elapsed = async {
1287            let start_instant = std::time::Instant::now();
1288            if migration.is_atomic() && !supports_transactional_ddl && !plan.fake {
1289                tracing::warn!(
1290                    "database does not support transactional DDL, atomic migration cannot be \
1291                     rolled back as a unit if it fails"
1292                );
1293            }
1294            match plan.plan_type {
1295                PlanType::Apply => {
1296                    tracing::info!("applying migration");
1297                    let operations = migration.operations();
1298                    if migration.is_atomic() {
1299                        let mut transaction = connection.begin().await?;
1300                        if !plan.fake {
1301                            for operation in operations {
1302                                operation.up(&mut transaction).await?;
1303                            }
1304                        }
1305                        migrator
1306                            .add_migration_to_db_table(&mut transaction, migration.as_ref())
1307                            .await?;
1308                        transaction.commit().await?;
1309                    } else {
1310                        if !plan.fake {
1311                            for operation in operations {
1312                                operation.up(connection).await?;
1313                            }
1314                        }
1315                        migrator
1316                            .add_migration_to_db_table(connection, migration.as_ref())
1317                            .await?;
1318                    }
1319                }
1320                PlanType::Revert => {
1321                    tracing::info!("reverting migration");
1322
1323                    // Reverse operation since last applied operation need
1324                    // to be reverted first
1325                    let mut operations = migration.operations();
1326                    operations.reverse();
1327
1328                    if migration.is_atomic() {
1329                        let mut transaction = connection.begin().await?;
1330                        if !plan.fake {
1331                            for operation in operations {
1332                                operation.down(&mut transaction).await?;
1333                            }
1334                        }
1335                        migrator
1336                            .delete_migration_from_db_table(&mut transaction, migration.as_ref())
1337                            .await?;
1338                        transaction.commit().await?;
1339                    } else {
1340                        if !plan.fake {
1341                            for operation in operations {
1342                                operation.down(connection).await?;
1343                            }
1344                        }
1345                        migrator
1346                            .delete_migration_from_db_table(connection, migration.as_ref())
1347                            .await?;
1348                    }
1349                }
1350            }
1351            let elapsed = start_instant.elapsed();
1352            tracing::info!(elapsed_ms = elapsed.as_millis(), "migration finished");
1353            Ok::<_, Error>(elapsed)
1354        }
1355        .instrument(migration_span)
1356        .await?;
1357        report.records.push(RunRecord {
1358            app: migration.app().to_string(),
1359            name: migration.name().to_string(),
1360            elapsed,
1361        });
1362    }
1363    Ok(report)
1364}

1369/// A struct that stores migration-related metadata, including the list of
1370/// migrations and configuration such as table and schema name

1373    table_prefix: Option<String>,
1374    schema: Option<String>,

1378impl<DB> Migrator<DB> {
1379    /// Creates a new migrator
1380    ///
1381    /// # Example
1382    /// ```rust
1383    /// # #[cfg(feature="sqlite")]
1384    /// # fn main() {
1385    /// let migrator = sqlx_migrator::Migrator::<sqlx::Sqlite>::new();
1386    /// assert_eq!(&migrator.table_name(), "_sqlx_migrator_migrations")
1387    /// # }
1388    /// # #[cfg(not(feature="sqlite"))]
1389    /// # fn main() {
1390    /// # }
1391    /// ```
1392    #[must_use]
1393    pub fn new() -> Self {
1394        Self {
1395            migrations: Vec::default(),
1396            table_prefix: None,
1397            schema: None,
1398            lock_strategy: LockStrategy::default(),
1399        }
1400    }
1401
1402    /// Configures how the migrator locks the database while running.
1403    ///
1404    /// Defaults to [`LockStrategy::Database`]. See [`LockStrategy`] for when
1405    /// another strategy is appropriate.
1406    ///
1407    /// # Example
1408    /// ```rust
1409    /// # #[cfg(feature="sqlite")]
1410    /// # fn main() {
1411    /// use sqlx_migrator::migrator::LockStrategy;
1412    /// let migrator =
1413    ///     sqlx_migrator::Migrator::<sqlx::Sqlite>::new().set_lock_strategy(LockStrategy::None);
1414    /// assert_eq!(migrator.lock_strategy(), LockStrategy::None);
1415    /// # }
1416    /// # #[cfg(not(feature="sqlite"))]
1417    /// # fn main() {}
1418    /// ```
1419    #[must_use]
1420    pub fn set_lock_strategy(mut self, strategy: LockStrategy) -> Self {
1421        self.lock_strategy = strategy;
1422        self
1423    }
1424
1425    /// Returns the configured lock strategy
1426    #[must_use]
1427    pub fn lock_strategy(&self) -> LockStrategy {
1428        self.lock_strategy
1429    }
1430
1431    /// Configures a prefix for the migrator table name.
1432    ///
1433    /// The table name will be formatted as
1434    /// `_{prefix}_sqlx_migrator_migrations`. Only ASCII lowercase, numeric
1435    /// characters and underscores are allowed in the prefix.
1436    ///
1437    /// # Example
1438    /// ```rust
1439    /// # #[cfg(feature="sqlite")]
1440    /// # fn main() {
1441    /// let migrator = sqlx_migrator::Migrator::<sqlx::Sqlite>::new()
1442    ///     .set_table_prefix("prefix_value")
1443    ///     .unwrap();
1444    /// assert_eq!(
1445    ///     &migrator.table_name(),
1446    ///     "_prefix_value_sqlx_migrator_migrations"
1447    /// )
1448    /// # }
1449    /// # #[cfg(not(feature="sqlite"))]
1450    /// # fn main() {
1451    /// # }
1452    /// ```
1453    ///
1454    /// # Errors
1455    /// When passed table prefix name contains invalid characters
1456    pub fn set_table_prefix(mut self, prefix: impl Into<String>) -> Result<Self, Error> {
1457        let prefix_str = prefix.into();
1458        if prefix_str.is_empty()
1459            || !prefix_str
1460                .chars()
1461                .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_')
1462        {
1463            return Err(Error::InvalidTablePrefix);
1464        }
1465        self.table_prefix = Some(prefix_str);
1466        Ok(self)
1467    }
1468
1469    /// Configures a schema for the migrator table.
1470    ///
1471    /// When set, the table name will be formatted as `{schema}.{table_name}`.
1472    /// Schema name can only contain [a-z0-9_] and begin with [a-z_]
1473    ///
1474    /// For Postgres the schema is created automatically (`CREATE SCHEMA IF
1475    /// NOT EXISTS`) when the migration table is ensured, so it does not need
1476    /// to exist beforehand. For MySQL the schema refers to a database which
1477    /// must already exist. SQLite has no schema support (other than attached
1478    /// databases), so setting a schema is not useful there.
1479    ///
1480    /// # Examples
1481    /// ```rust
1482    /// # #[cfg(feature="sqlite")]
1483    /// # fn main() {
1484    /// let migrator = sqlx_migrator::Migrator::<sqlx::Sqlite>::new()
1485    ///     .set_schema("migrations")
1486    ///     .unwrap();
1487    /// assert_eq!(
1488    ///     &migrator.table_name(),
1489    ///     "migrations._sqlx_migrator_migrations"
1490    /// );
1491    /// # }
1492    /// # #[cfg(not(feature="sqlite"))]
1493    /// # fn main() {}
1494    /// ```
1495    ///
1496    /// # Errors
1497    /// When passed schema name contains invalid characters
1498    pub fn set_schema(mut self, schema: impl Into<String>) -> Result<Self, Error> {
1499        let schema_str = schema.into();
1500        if schema_str.is_empty()
1501            || !schema_str
1502                .chars()
1503                .next()
1504                .is_some_and(|c| c.is_ascii_lowercase() || c == '_')
1505            || !schema_str
1506                .chars()
1507                .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_')
1508        {
1509            return Err(Error::InvalidSchema);
1510        }
1511        self.schema = Some(schema_str);
1512        Ok(self)
1513    }
1514
1515    /// Get name of table which is used for storing migrations related
1516    /// information in database
1517    ///
1518    /// Format depends on configuration:
1519    /// - With schema: `{schema}._sqlx_migrator_migrations`
1520    /// - With prefix: `_{prefix}_sqlx_migrator_migrations`
1521    /// - With both: `{schema}._{prefix}_sqlx_migrator_migrations`
1522    /// - Default: `_sqlx_migrator_migrations`
1523    ///
1524    /// # Examples
1525    /// ```rust
1526    /// # #[cfg(feature="sqlite")]
1527    /// # fn main() {
1528    /// let migrator = sqlx_migrator::Migrator::<sqlx::Sqlite>::new()
1529    ///     .set_schema("app_schema")
1530    ///     .unwrap()
1531    ///     .set_table_prefix("v1")
1532    ///     .unwrap();
1533    /// assert_eq!(
1534    ///     &migrator.table_name(),
1535    ///     "app_schema._v1_sqlx_migrator_migrations"
1536    /// );
1537    /// # }
1538    /// # #[cfg(not(feature="sqlite"))]
1539    /// # fn main() {}
1540    /// ```
1541    #[must_use]
1542    pub fn table_name(&self) -> String {
1543        self.qualified_name(DEFAULT_TABLE_NAME)
1544    }
1545
1546    /// Get name of the meta table which records the migrator's internal
1547    /// migrations. It follows the same prefix and schema rules as
1548    /// [`Migrator::table_name`].
1549    ///
1550    /// # Examples
1551    /// ```rust
1552    /// # #[cfg(feature="sqlite")]
1553    /// # fn main() {
1554    /// let migrator = sqlx_migrator::Migrator::<sqlx::Sqlite>::new()
1555    ///     .set_table_prefix("v1")
1556    ///     .unwrap();
1557    /// assert_eq!(&migrator.meta_table_name(), "_v1_sqlx_migrator_meta");
1558    /// # }
1559    /// # #[cfg(not(feature="sqlite"))]
1560    /// # fn main() {}
1561    /// ```
1562    #[must_use]
1563    pub fn meta_table_name(&self) -> String {
1564        self.qualified_name(DEFAULT_META_TABLE_NAME)
1565    }
1566
1567    /// Applies the configured prefix and schema to a base table name
1568    fn qualified_name(&self, base: &str) -> String {
1569        let mut table_name = base.to_string();
1570        if let Some(prefix) = &self.table_prefix {
1571            table_name = format!("_{prefix}{table_name}");
1572        }
1573        if let Some(schema) = &self.schema {
1574            table_name = format!("{schema}.{table_name}");
1575        }
1576        table_name
1577    }
1578}
