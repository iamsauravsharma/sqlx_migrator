1//! The migrations of a migrator as a graph, which orders them and narrows them
2//! down to what a plan asks for.
3//!
4//! Parents, run before and replaces of every migration are read once when the
5//! graph is built, everything after that works on indexes into its nodes.
6
7use std::collections::{BTreeSet, HashMap, HashSet, VecDeque};
8
9use super::{BoxMigration, Plan, PlanType};
10use crate::error::Error;
11
12/// A migration and how it relates to the others, which are given as indexes
13/// into the nodes, that is into the migration list.
14struct Node {
15    app: String,
16    name: String,
17    /// Migrations it has to run after
18    parents: Vec<usize>,
19    /// Migrations which said they run before it, so it runs after them too
20    run_after: Vec<usize>,
21    /// Migrations it said it runs before
22    run_before: Vec<usize>,
23    /// The migration replacing it
24    replaced_by: Option<usize>,
25    /// Migrations it replaces
26    replaces: Vec<usize>,
27    /// Migrations it replaces and the ones they replace, and so on
28    all_replaces: Vec<usize>,
29}
30
31impl Node {
32    fn new(app: &str, name: &str) -> Self {
33        Self {
34            app: app.to_string(),
35            name: name.to_string(),
36            parents: Vec::new(),
37            run_after: Vec::new(),
38            run_before: Vec::new(),
39            replaced_by: None,
40            replaces: Vec::new(),
41            all_replaces: Vec::new(),
42        }
43    }
44}

46/// How the migrations of the migration list relate to each other, one node
47/// for each migration in the order of the list
48pub(super) struct Relations {
49    nodes: Vec<Node>,
50}
51
52impl Relations {
53    /// Graph of the migrations in the migration list
54    ///
55    /// # Errors
56    /// When a migration refers to one which is not in the list, is replaced
57    /// twice or is replaced by a migration it replaces
58    pub(super) fn new<DB>(migrations: &[BoxMigration<DB>]) -> Result<Self, Error> {
59        let index_of: HashMap<(&str, &str), usize> = migrations
60            .iter()
61            .enumerate()
62            .map(|(index, migration)| ((migration.app(), migration.name()), index))
63            .collect();
64        let index = |migration: &BoxMigration<DB>| {
65            index_of
66                .get(&(migration.app(), migration.name()))
67                .copied()
68                .ok_or_else(|| {
69                    Error::PlanError {
70                        message: format!(
71                            "migration {}:{} is not in the migration list",
72                            migration.app(),
73                            migration.name()
74                        ),
75                    }
76                })
77        };
78        let mut nodes = migrations
79            .iter()
80            .map(|migration| Node::new(migration.app(), migration.name()))
81            .collect::<Vec<_>>();
82        for (migration_index, migration) in migrations.iter().enumerate() {
83            nodes[migration_index].parents = migration
84                .parents()
85                .iter()
86                .map(index)
87                .collect::<Result<_, _>>()?;
88            for target in migration.run_before() {
89                let target = index(&target)?;
90                nodes[target].run_after.push(migration_index);
91                nodes[migration_index].run_before.push(target);
92            }
93        }
94        // a migration can only be replaced by one migration but one migration
95        // can replace many
96        for (migration_index, migration) in migrations.iter().enumerate() {
97            for child in migration.replaces() {
98                let child_index = index(&child)?;
99                if nodes[child_index].replaced_by.is_some() {
100                    return Err(Error::PlanError {
101                        message: format!(
102                            "migration {}:{} replaced multiple times",
103                            child.app(),
104                            child.name()
105                        ),
106                    });
107                }
108                nodes[child_index].replaced_by = Some(migration_index);
109                nodes[migration_index].replaces.push(child_index);
110            }
111        }
112        let mut relations = Self { nodes };
113        for replacer in 0..relations.nodes.len() {
114            relations.nodes[replacer].all_replaces = relations.find_all_replaces(replacer)?;
115        }
116        Ok(relations)
117    }

119    /// Migrations a migration replaces and the ones they replace
120    fn find_all_replaces(&self, replacer: usize) -> Result<Vec<usize>, Error> {
121        let mut all = Vec::new();
122        let mut seen = HashSet::new();
123        let mut stack = self.nodes[replacer].replaces.clone();
124        while let Some(migration) = stack.pop() {
125            if migration == replacer {
126                return Err(Error::PlanError {
127                    message: "two migrations replace each other".to_string(),
128                });
129            }
130            if seen.insert(migration) {
131                all.push(migration);
132                stack.extend(&self.nodes[migration].replaces);
133            }
134        }
135        Ok(all)
136    }

138    /// Order of the migrations in which they can be applied.
139    ///
140    /// A migration comes after its parents, the migrations which have it in
141    /// their run before and the migration replacing it, and a migration
142    /// replacing others also after the parents and run before of those.
143    /// Of the migrations which are free the first one of the list is added
144    /// first.
145    ///
146    /// # Errors
147    /// When the migrations wait for each other
148    pub(super) fn order(&self) -> Result<Vec<usize>, Error> {
149        let mut waiting_for = vec![0_usize; self.nodes.len()];
150        let mut dependents = vec![Vec::new(); self.nodes.len()];
151        for migration in 0..self.nodes.len() {
152            let node = &self.nodes[migration];
153            let mut prerequisites = node
154                .parents
155                .iter()
156                .chain(&node.run_after)
157                .copied()
158                .collect::<HashSet<_>>();
159            prerequisites.extend(node.replaced_by);
160            let replaced = node.all_replaces.iter().collect::<HashSet<_>>();
161            for &child in &node.all_replaces {
162                prerequisites.extend(&self.nodes[child].parents);
163                prerequisites.extend(
164                    self.nodes[child]
165                        .run_after
166                        .iter()
167                        .filter(|parent| !replaced.contains(parent)),
168                );
169            }
170            waiting_for[migration] = prerequisites.len();
171            for before in prerequisites {
172                dependents[before].push(migration);
173            }
174        }
175
176        let mut free = (0..self.nodes.len())
177            .filter(|&migration| waiting_for[migration] == 0)
178            .collect::<BTreeSet<_>>();
179        let mut order = Vec::with_capacity(self.nodes.len());
180        while let Some(next) = free.pop_first() {
181            order.push(next);
182            for &dependent in &dependents[next] {
183                waiting_for[dependent] -= 1;
184                if waiting_for[dependent] == 0 {
185                    free.insert(dependent);
186                }
187            }
188        }
189        // migrations which are left wait for each other
190        if order.len() != self.nodes.len() {
191            return Err(Error::PlanError {
192                message: "reached deadlock stage during plan generation".to_string(),
193            });
194        }
195        Ok(order)
196    }

198    /// Narrows the order of all migrations down to what `plan` asks for,
199    /// given which migrations are applied
200    ///
201    /// # Errors
202    /// When the applied migrations do not make sense together or the plan
203    /// names an app, a migration or a count which does not exist
204    pub(super) fn plan(
205        &self,
206        mut order: Vec<usize>,
207        applied: &[bool],
208        plan: &Plan,
209        last_batch_keys: &HashSet<(&str, &str)>,
210    ) -> Result<Vec<usize>, Error> {
211        self.check_applied(applied)?;
212        self.remove_replaced(&mut order, applied)?;
213        self.process(&mut order, applied, plan, last_batch_keys)?;
214        Ok(order)
215    }
216
217    /// Checks that no applied migration comes before one of its parents, the
218    /// parent being applied itself or through a migration replacing it
219    fn check_applied(&self, applied: &[bool]) -> Result<(), Error> {
220        for migration in (0..self.nodes.len()).filter(|&migration| applied[migration]) {
221            let parents = self.nodes[migration]
222                .run_after
223                .iter()
224                .chain(&self.nodes[migration].parents);
225            for &parent in parents {
226                // the parent itself or a migration replacing it
227                let satisfied = std::iter::successors(Some(parent), |&candidate| {
228                    self.nodes[candidate].replaced_by
229                })
230                .any(|candidate| applied[candidate]);
231                if !satisfied {
232                    return Err(Error::PlanError {
233                        message: format!(
234                            "child migration {}:{} applied before its parent migration {}:{}",
235                            self.nodes[migration].app,
236                            self.nodes[migration].name,
237                            self.nodes[parent].app,
238                            self.nodes[parent].name
239                        ),
240                    });
241                }
242            }
243        }
244        Ok(())
245    }

247    /// Removes the migrations which are replaced and the ones replacing a
248    /// migration which is applied, since one of them stands in for the other
249    fn remove_replaced(&self, order: &mut Vec<usize>, applied: &[bool]) -> Result<(), Error> {
250        let mut remove = vec![false; self.nodes.len()];
251        for &migration in order.iter() {
252            let replaced = &self.nodes[migration].all_replaces;
253            if replaced.iter().any(|&child| applied[child]) {
254                // both cannot be applied together
255                if applied[migration] {
256                    return Err(Error::PlanError {
257                        message: format!(
258                            "migration {}:{} and its replaces are applied together",
259                            self.nodes[migration].app, self.nodes[migration].name,
260                        ),
261                    });
262                }
263                remove[migration] = true;
264            } else {
265                // this migration applies in place of them
266                for &child in replaced {
267                    remove[child] = true;
268                }
269            }
270        }
271        order.retain(|&migration| !remove[migration]);
272        Ok(())
273    }

275    /// Keeps the migrations of `order` the plan asks for
276    fn process(
277        &self,
278        order: &mut Vec<usize>,
279        applied: &[bool],
280        plan: &Plan,
281        last_batch_keys: &HashSet<(&str, &str)>,
282    ) -> Result<(), Error> {
283        match &plan.plan_type {
284            PlanType::Apply => order.retain(|&migration| !applied[migration]),
285            PlanType::Revert(revert_plan) => {
286                order.retain(|&migration| applied[migration]);
287                // If the revert plan targets the last batch, filter out
288                // migrations not in the last batch.
289                if revert_plan.last_batch {
290                    order.retain(|&migration| {
291                        last_batch_keys
292                            .contains(&(&self.nodes[migration].app, &self.nodes[migration].name))
293                    });
294                }
295                order.reverse();
296            }
297        }
298
299        if let Some((app, migration_name)) = &plan.app_migration {
300            // A target that exists in the full migration list but not in the
301            // processed list has nothing to do (it is already applied/reverted
302            // or replaced), so the plan is empty rather than an error. This
303            // keeps targeted apply/revert idempotent. Only a target that
304            // doesn't exist at all is an error.
305            let is_target = |migration: usize| {
306                self.nodes[migration].app == *app
307                    && migration_name
308                        .as_ref()
309                        .is_none_or(|name| self.nodes[migration].name == *name)
310            };
311            let Some(position) = order.iter().rposition(|&migration| is_target(migration)) else {
312                if !(0..self.nodes.len()).any(|migration| self.nodes[migration].app == *app) {
313                    return Err(Error::PlanError {
314                        message: format!("app {app} doesn't exist"),
315                    });
316                }
317                if let Some(name) = migration_name
318                    && !(0..self.nodes.len()).any(is_target)
319                {
320                    return Err(Error::PlanError {
321                        message: format!("migration {name} doesn't exist for app {app}"),
322                    });
323                }
324                order.clear();
325                return Ok(());
326            };
327            order.truncate(position + 1);
328            let targets = order
329                .iter()
330                .copied()
331                .filter(|&migration| is_target(migration))
332                .collect::<Vec<_>>();
333            self.only_related(order, &targets, &plan.plan_type);
334        } else if let Some(count) = plan.count {
335            if count == 0 {
336                return Err(Error::PlanError {
337                    message: "count must be greater than 0".to_string(),
338                });
339            }
340            let actual_len = order.len();
341            if count > actual_len {
342                return Err(Error::PlanError {
343                    message: format!(
344                        "passed count value is larger than migration length: {actual_len}"
345                    ),
346                });
347            }
348            order.truncate(count);
349        }
350        Ok(())
351    }

353    /// Narrows `order` down to the migrations required to apply or revert the
354    /// `targets`, which are the ones they depend on when applying and the ones
355    /// depending on them when reverting. A migration replacing others also
356    /// takes over the relations of those.
357    fn only_related(&self, order: &mut Vec<usize>, targets: &[usize], plan_type: &PlanType) {
358        let mut in_order = vec![false; self.nodes.len()];
359        for &migration in order.iter() {
360            in_order[migration] = true;
361        }
362        let neighbours = match plan_type {
363            PlanType::Apply => self.dependencies(),
364            PlanType::Revert(_) => self.dependents(),
365        };
366        let mut related = vec![false; self.nodes.len()];
367        let mut queue = targets.iter().copied().collect::<VecDeque<_>>();
368        for &target in targets {
369            related[target] = true;
370        }
371        while let Some(current) = queue.pop_front() {
372            for &migration in &neighbours[current] {
373                if in_order[migration] && !related[migration] {
374                    related[migration] = true;
375                    queue.push_back(migration);
376                }
377            }
378        }
379        order.retain(|&migration| related[migration]);
380    }

382    /// For every migration, the migrations which have to come before it:
383    /// its parents and the migrations having it in their run before, and the
384    /// same of the migrations it replaces
385    fn dependencies(&self) -> Vec<Vec<usize>> {
386        (0..self.nodes.len())
387            .map(|migration| {
388                std::iter::once(migration)
389                    .chain(self.nodes[migration].all_replaces.iter().copied())
390                    .flat_map(|source| {
391                        self.nodes[source]
392                            .parents
393                            .iter()
394                            .chain(&self.nodes[source].run_after)
395                            .copied()
396                    })
397                    .collect()
398            })
399            .collect()
400    }

402    /// For every migration, the migrations which have to come after it: the
403    /// ones with it as a parent, directly or through a migration they replace,
404    /// and the ones with a migration it has to come before, or one that
405    /// migration replaces, in their run before... read the other way round
406    fn dependents(&self) -> Vec<Vec<usize>> {
407        let mut dependents = vec![Vec::new(); self.nodes.len()];
408        // migrations named by the run before of others, and the migrations
409        // replacing them, which are all reverted after the others
410        let mut replacing = vec![Vec::new(); self.nodes.len()];
411        for (migration, node) in self.nodes.iter().enumerate() {
412            for source in std::iter::once(migration).chain(node.all_replaces.iter().copied()) {
413                for &parent in &self.nodes[source].parents {
414                    dependents[parent].push(migration);
415                }
416                replacing[source].push(migration);
417            }
418        }
419        for (migration, node) in self.nodes.iter().enumerate() {
420            for &target in &node.run_before {
421                dependents[migration].extend(&replacing[target]);
422            }
423        }
424        dependents
425    }
