3/// Error enum to store different types of error
4#[derive(Debug, thiserror::Error)]
5#[non_exhaustive]
6pub enum Error {
7    /// Error type created from error raised by sqlx
8    #[error(transparent)]
9    Sqlx(#[from] sqlx::Error),
10    /// Error type created from error raised by box error
11    #[error(transparent)]
12    Box(#[from] Box<dyn std::error::Error + Send + Sync>),
13    /// Error type created from error raised by std input output
14    #[cfg(feature = "cli")]
15    #[error(transparent)]
16    StdIo(#[from] std::io::Error),
17    /// Error type created when cli output cannot be serialized as JSON
18    #[cfg(feature = "json")]
19    #[error(transparent)]
20    Json(#[from] serde_json::Error),
21    /// Error generated during planning state
22    #[error("plan error: {message}")]
23    PlanError {
24        /// Message for error
25        message: String,
26    },
27    /// Error for irreversible operation
28    #[error("operation is irreversible")]
29    IrreversibleOperation,
30    /// Error for pending migration present
31    #[error("pending migrations exists. Apply pending migrations first")]
32    PendingMigrationPresent,
33    /// Error when applied migrations exists
34    #[cfg(feature = "cli")]
35    #[error("applied migrations exists. Revert all using revert subcommand")]
36    AppliedMigrationExists,
37    /// Error when confirmation is required but stdin is not a terminal so no
38    /// confirmation can be asked
39    #[cfg(feature = "cli")]
40    #[error(
41        "confirmation is required but stdin is not a terminal. Use --force flag to run \
42         non-interactively"
43    )]
44    ConfirmationRequired,
45    /// Error when table prefix is invalid
46    #[error("table prefix name can only contain [a-z0-9_]")]
47    InvalidTablePrefix,
48    /// Error when passed schema name is invalid
49    #[error("schema name can only contain [a-z0-9_] and begin with [a-z_]")]
50    InvalidSchema,
51    /// Error raised when two migration with same name are added and there value
52    /// is not consistent
53    #[error("inconsistent migration found for {app} - {name}")]
54    InconsistentMigration {
55        /// Migration application name
56        app: String,
57        /// Migration name
58        name: String,
59    },
60    /// Error raised when virtual migration is invalid. Virtual migration is
61    /// invalid if it has any fields present except app name and migration name
62    #[error("invalid virtual migration")]
63    InvalidVirtualMigration,
64    /// Error raised when the database does not have the lock of a lock
65    /// strategy
66    #[error("lock strategy {strategy:?} is not supported by this database")]
67    UnsupportedLockStrategy {
68        /// The strategy which was asked for
69        strategy: crate::migrator::LockStrategy,
70    },
71    /// Error raised when the lock row of the lock table is held by another run
72    #[error(
73        "migration lock in {table} is held by another run. If a previous run crashed, delete its \
74         row from the lock table"
75    )]
76    LockHeld {
77        /// Name of the lock table
78        table: String,
79    },
80    /// Error raised when a non atomic migration started running and never
81    /// finished, so the database may be partly migrated
82    #[error(
83        "migration {app} - {name} failed and left the database partly migrated. Fix the database \
84         by hand and repair the migration table"
85    )]
86    FailedMigration {
87        /// Migration application name
88        app: String,
89        /// Migration name
90        name: String,
91    },
92    /// Error raised when the meta table records an internal migration this
93    /// version of the crate does not know, meaning the migration table was
94    /// changed by a newer release
95    #[error(
96        "migration table was changed by internal migration {name} which this version of \
97         sqlx_migrator does not know, upgrade sqlx_migrator"
98    )]
99    UnknownInternalMigration {
100        /// Name of the internal migration recorded in the database
101        name: String,
102    },
103}
